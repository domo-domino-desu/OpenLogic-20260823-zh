\olsection{语义概念}

我们定义以下语义概念：

\begin{defn}
\begin{enumerate}
\item 公式~$!A$ 是\emph{可满足的}，如果存在某个赋值~$\pAssign{v}$ 使得 $\pSat{v}{!A}$；如果不存在赋值~$\pAssign{v}$ 使得 $\pSat{v}{!A}$，则它是\emph{不可满足的}；
\item 公式~$!A$ 是\emph{重言式}，如果对所有赋值~$\pAssign{v}$，$\pSat{v}{!A}$；
\item 公式~$!A$ 是\emph{偶真的}，如果它可满足但不是重言式；
\item 如果 $\Gamma$ 是公式集合，则 $\Gamma \Entails !A$（“$\Gamma$ 语义地后承 $!A$”）当且仅当对于每个满足 $\pSat{v}{\Gamma}$ 的赋值~$\pAssign{v}$，$\pSat{v}{!A}$。
\item 如果 $\Gamma$ 是公式集合，则 $\Gamma$ 是\emph{可满足的}，如果存在赋值~$\pAssign{v}$ 使得 $\pSat{v}{\Gamma}$；否则 $\Gamma$ 是\emph{不可满足的}。
\end{enumerate}
\end{defn}

\begin{prob}
对以下四个公式，分别判断它们是否为 ~可满足的，~重言式，以及 ~偶真的。
\begin{enumerate}
  \item \( ( \Obj p_0 \lif ( \lnot \Obj p_1 \lif \lnot \Obj p_0 ) ) \)。
  \item \( ( ( \Obj p_0 \land \lnot \Obj p_1 ) \lif ( \lnot \Obj p_0 \land \Obj p_2 )) \liff ( ( \Obj p_2 \lif \Obj p_0 ) \lif ( \Obj p_0 \lif \Obj p_1 )) \)。
  \item \( ( \Obj p_0 \liff \Obj p_1 ) \lif ( \Obj p_2 \liff \lnot \Obj p_1 ) \)。
  \item \( (( \Obj p_0 \liff ( \lnot \Obj p_1 \land \Obj p_2 )) \lor ( \Obj p_2 \lif ( \Obj p_0 \liff \Obj p_1 ))) \)。
\end{enumerate}
\end{prob}

\begin{prop}
\ollabel{prop:semanticalfacts}
\begin{enumerate}
\item $!A$ 是重言式当且仅当 $\emptyset \Entails !A$；
\item 如果 $\Gamma \Entails !A$ 且 $\Gamma \Entails !A \lif !B$，那么 $\Gamma \Entails !B$；
\item 如果 $\Gamma$ 是可满足的，那么 $\Gamma$ 的每个有限子集也是可满足的；
\item \ollabel{def:monotonicity} 单调性：如果 $\Gamma \subseteq \Delta$ 且 $\Gamma \Entails !A$，那么也有 $\Delta \Entails !A$；
\item \ollabel{def:Cut} 传递性：如果 $\Gamma \Entails !A$ 且 $\Delta \cup \{ !A\} \Entails !B$，那么 $\Gamma \cup \Delta \Entails !B$。
\end{enumerate}
\end{prop}

\begin{proof}
练习。
\end{proof}

\begin{prob}
证明 \olref[pl][syn][sem]{prop:semanticalfacts}。
\end{prob}

\begin{prop}\ollabel{prop:entails-unsat}
  $\Gamma \Entails !A$ 当且仅当 $\Gamma \cup \{\lnot !A\}$ 是不可满足的。
\end{prop}

\begin{proof}
练习。
\end{proof}

\begin{prob}
证明 \olref[pl][syn][sem]{prop:entails-unsat}。
\end{prob}

\begin{thm}[语义演绎定理]
  \ollabel{thm:sem-deduction} $\Gamma \Entails !A \lif !B$ 当且仅当 $\Gamma \cup \{!A\} \Entails !B$。
\end{thm}

\begin{proof}
练习。
\end{proof}

\begin{prob}
证明 \olref[pl][syn][sem]{thm:sem-deduction}。
\end{prob}
