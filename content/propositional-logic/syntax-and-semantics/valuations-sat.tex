\olsection{赋值与满足}

\begin{defn}[赋值]
设 $\{\True, \False\}$ 为由“真”与“假”两个真值组成的集合。
$\Lang{L_0}$ 的一个\emph{赋值}是一个函数~$\pAssign{v}$，
它为语言中的每个命题变元指派 $\True$ 或 $\False$ 之一，
即 $\pAssign{v} \colon
\PVar \to \{\True, \False \}$。
\end{defn}

\begin{defn}
  给定一个赋值~$\pAssign{v}$，归纳地定义求值函数
  $\pValue{v} \colon \Frm[L_0] \to \{\True, \False \}$ 如下：
  \begin{align*}
    \iftag{prvFalse}{\pValue{v}(\lfalse) & = \False; \\}{}
    \iftag{prvTrue}{\pValue{v}(\ltrue) & = \True; \\}{}
    \pValue{v}(\Obj p_n) & = \pAssign{v}(\Obj p_n); \\
    \iftag{prvNot}{\pValue{v}(\lnot !A) & = \begin{cases}
        \True & \text{若 } \pValue{v}(!A) = \False;\\
        \False & \text{否则。}
      \end{cases} \\ }{}
    \iftag{prvAnd}{\pValue{v}(!A \land !B) & = \begin{cases}
        \True &
        \text{若 $\pValue{v}(!A) = \True$ 且 $\pValue{v}(!B) = \True$；}\\
        \False &
        \text{若 $\pValue{v}(!A) = \False$ 或 $\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvOr}{\pValue{v}(!A \lor !B) & = \begin{cases}
        \True &
        \text{若 $\pValue{v}(!A) = \True$ 或 $\pValue{v}(!B) = \True$；}\\
        \False &
        \text{若 $\pValue{v}(!A) = \False$ 且 $\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvIf}{\pValue{v}(!A \lif !B) & = \begin{cases}
        \True &
        \text{若 $\pValue{v}(!A) = \False$ 或 $\pValue{v}(!B) = \True$；}\\
        \False &
        \text{若 $\pValue{v}(!A) = \True$ 且 $\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvIff}{\pValue{v}(!A \liff !B) & = \begin{cases}
        \True &
        \text{若 $\pValue{v}(!A) = \pValue{v}(!B)$；}\\
        \False &
        \text{若 $\pValue{v}(!A) \neq \pValue{v}(!B)$}.
      \end{cases}}{}
\end{align*}
\end{defn}

\begin{explain}
这些条款对应于以下真值表：
\begin{center}
\iftag{prvNot}{
  \begin{tabular}{|c||c|} \hline
    $!A$ & $ \lnot !A$ \\
    \hline \hline
    $\True$ & $\False$ \\
    $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvAnd}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \land !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\False$ \\
    $\False$ & $\False$ & $\False$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvOr}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \lor !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\True$ \\
    $\False$ & $\True$ & $\True$ \\
    $\False$ & $\False$ & $\False$ \\
    \hline
  \end{tabular}
}{}%
\iftag{notprvNot,notprvAnd,notprvOr,notprvIf}{}{\\[1em]}
\iftag{prvIf}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \lif !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\True$ \\
    $\False$ & $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvIff}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \liff !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\False$ \\
    $\False$ & $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\end{center}
\end{explain}

\begin{prob}
考虑在 $\Lang{L_0}$ 中添加一个三元联结词 $\diamondsuit$，其求值由下式给出：
\begin{gather*}
  \pValue{v}(\diamondsuit( !A , !B , !C ) ) = \begin{cases}
    \pValue{v}( !B ) &
    \text{若 $\pValue{v}(!A) = \True$；}\\
    \pValue{v}( !C ) &
    \text{若 $\pValue{v}(!A) = \False $}.
  \end{cases}
\end{gather*}
试写出这个联结词的真值表。
\end{prob}

\begin{thm}[局部决定性]
\ollabel{thm:LocalDetermination} 假设 $\pAssign{v_1}$ 和
$\pAssign{v_2}$ 是两个赋值，并且它们在公式 $!A$ 中出现的所有命题变元上一致，
即若 $\Obj p_n$ 在公式~$!A$ 中出现，则 $\pAssign{v_1}(\Obj p_n) =
\pAssign{v_2}(\Obj p_n)$。
那么 $\pValue{v_1}$ 和 $\pValue{v_2}$ 在~$!A$ 上也一致，
即 $\pValue{v_1}(!A) = \pValue{v_2}(!A)$。
\end{thm}

\begin{proof}
对 $!A$ 施归纳。
\end{proof}

\begin{defn}[满足]
\ollabel{defn:satisfaction} 我们可以如下归纳地定义
  \emph{公式~$!A$ 被赋值~$\pAssign{v}$ 满足}的概念，
  记作 $\pSat{v}{!A}$：
  （我们用 $\pSat/{v}{!A}$ 表示“并非 $\pSat{v}{!A}$”。）
\begin{enumerate}
\tagitem{prvFalse}{%
  \indcase{!A}{\lfalse}{$\pSat/{v}{\indfrm}$.}}{}

\tagitem{prvTrue}{%
  \indcase{!A}{\ltrue}{$\pSat{v}{\indfrm}$.}}{}

\item \indcase{!A}{\Obj p_i}{$\pSat{v}{\indfrm}$
  当且仅当 $\pAssign{v}(\Obj p_i) = \True$。}

\tagitem{prvNot}{%
  \indcase{!A}{\lnot !B}{$\pSat{v}{\indfrm}$ 当且仅当
    $\pSat/{v}{!B}$。}}{}

\tagitem{prvAnd}{%
  \indcase{!A}{(!B \land !C)}{$\pSat{v}{\indfrm}$ 当且仅当 $\pSat{v}{!B}$
    且 $\pSat{v}{!C}$。}}{}

\tagitem{prvOr}{%
  \indcase{!A}{(!B \lor !C)}{$\pSat{v}{\indfrm}$ 当且仅当
    $\pSat{v}{!B}$ 或 $\pSat{v}{!C}$（或两者均成立）。}}{}

\tagitem{prvIf}{%
  \indcase{!A}{(!B \lif !C)}{$\pSat{v}{\indfrm}$ 当且仅当 $\pSat/{v}{!B}$
    或 $\pSat{v}{!C}$（或两者均成立）。}}{}

\tagitem{prvIff}{%
  \indcase{!A}{(!B \liff !C)}{$\pSat{v}{\indfrm}$ 当且仅当要么
    $\pSat{v}{!B}$ 与 $\pSat{v}{!C}$ 均成立，要么两者均不成立。}}{}
\end{enumerate}
若 $\Gamma$ 是公式集合，则 $\pSat{v}{\Gamma}$ 当且仅当对每个~$!A \in \Gamma$ 都有 $\pSat{v}{!A}$。
\end{defn}

\begin{prop}\ollabel{prop:sat-value}
  $\pSat{v}{!A}$ 当且仅当 $\pValue{v}(!A) = \True$。
\end{prop}

\begin{proof}
  对~$!A$ 施归纳。
\end{proof}

\begin{prob}
  证明 \olref[pl][syn][val]{prop:sat-value}。
\end{prob}
