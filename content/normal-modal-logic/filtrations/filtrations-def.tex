\olsection{过滤}

我们不去定义 $\mModel{M}$ 通过 $\Gamma$ 的“那个”过滤，而是定义模型~$\mModel{M^*}$ 何时算作~$\mModel{M}$ 的一个过滤。
所有过滤都具有相同的世界集~$W^*$ 和赋值~$V^*$。
但不同的过滤可能有不同的可达关系~$R^*$。
然而，要算作过滤，$R^*$ 必须满足若干条件。
这些条件正是我们证明主要结果所需要的，即：如果 $!A \in \Gamma$，则 $\mSat{M}{!A}[w]$ 当且仅当 $\mSat{M^*}{!A}[{[w]}]$。

\begin{defn}\ollabel{defn:filtration}
  设 $\Gamma$ 对子公式封闭且包含 $\mModel{M} = \tuple{W,
    R, V}$。
  $\mModel{M}$ 通过 $\Gamma$ 的一个\emph{过滤}是任意满足以下条件的模型 $\mModel{M^*} = \tuple{W^*,R^*,V^*}$：
  \begin{enumerate}
  \item $W^* = \Setabs{[w]}{w \in W}$；
  \item \ollabel{defn:filtration-R}%
    对任意 $u,v \in W$：
    \begin{enumerate}
    \item \ollabel{defn:filtration-R1}%
      若 $Ruv$，则 $R^*[u][v]$；
    \tagitem{prvBox}{\ollabel{defn:filtration-R2}%
      若 $R^*[u][v]$，则对任意 $\Box !A \in \Gamma$，
      若 $\mSat{M}{\Box!A}[u]$，则 $\mSat{M}{!A}[v]$；}{}
    \tagitem{prvDiamond}{\ollabel{defn:filtration-R3}%
      若 $R^*[u][v]$，则对任意 $\Diamond !A \in \Gamma$，
      若 $\mSat{M}{!A}[v]$，则 $\mSat{M}{\Diamond!A}[u]$。}{}
    \end{enumerate}
  \item $V^*(p) = \Setabs{[u]}{u \in V(p)}$。
  \end{enumerate}
\end{defn}

思考一下 $V^*(p)$ 是什么是有益的：它是由所有在~$\mModel{M}$ 中 $p$ 为真的世界~$w$ 的等价类~$[w]$ 构成的集合。
如果 $w \in V(p)$，则根据定义有 $[w] \in V^*(p)$。
然而，若 $[w] \in V^*(p)$，未必有 $w \in V(p)$。
若 $[w] \in V^*(p)$，我们只能保证存在\emph{某个} $u \in V(p)$ 使得 $[w] = [u]$。
当然，$[w] = [u]$ 意味着 $w \equiv u$。
所以，当 $[w] \in V^*(p)$ 时，我们（仅）能断定存在某个 $u \in V(p)$ 使得 $w \equiv u$。

\begin{thm}\ollabel{thm:filtrations}
  若 $\mModel{M^*}$ 是 $\mModel{M}$ 通过 $\Gamma$ 的一个过滤，
  则对每个 $!A \in \Gamma$ 和 $w \in W$，有
  $\mSat{M}{!A}[w]$ 当且仅当 $\mSat{M^*}{!A}[{[w]}]$。
\end{thm}

\begin{proof}
  对 $!A$ 施加归纳，利用 $\Gamma$ 对子公式封闭这一事实。因为 $!A \in \Gamma$ 且 $\Gamma$ 对子公式封闭，$!A$ 的所有子公式也属于 $\Gamma$。因此在每个归纳步骤中，归纳假设均适用于 $!A$ 的子公式。
  \begin{enumerate}
    \tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\mSat{M}{\indfrm}[w]$ 和 $\mSat{M^*}{\indfrm}[{[w]}]$ 均不成立。}}{}
    \tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\mSat{M}{\indfrm}[w]$ 和 $\mSat{M^*}{\indfrm}[{[w]}]$ 均成立。}}{}
    \item \indcase{!A}{p}{从左到右的方向是直接的，因为
      $\mSat{M}{\indfrm}[w]$ 仅当 $w \in V(p)$，而这蕴含 $[w]
      \in V^*(p)$，即 $\mSat{M^*}{\indfrm}[{[w]}]$。反之，
      假设 $\mSat{M^*}{\indfrm}[{[w]}]$，即 $[w] \in
      V^*(p)$。则存在某个 $v \in V(p)$ 使得 $w \equiv v$。显然此时也有 $\mSat{M}{p}[v]$。由于 $w \equiv v$，$w$ 和 $v$ 使得 $\Gamma$ 中相同的公式为真。因为根据假设 $p
      \in \Gamma$ 且 $\mSat{M}{p}[v]$，故有 $\mSat{M}{\indfrm}[w]$。}

    \tagitem{prvNot}{\iftag{probNot}{练习。}{\indcase{!A}{\lnot
          !B}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat/{M}{!B}[w]$。由
          归纳假设，$\mSat/{M}{!B}[w]$ 当且仅当
          $\mSat/{M^*}{!B}[{[w]}]$。 最后，$\mSat/{M^*}{!B}[{[w]}]$
          当且仅当 $\mSat{M^*}{\indfrm}[{[w]}]$。}}}{}
        
    \tagitem{prvAnd}{\iftag{probAnd}{练习。}{\indcase{!A}{(!B \land
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat{M}{!B}[w]$ 且
          $\mSat{M}{!C}[w]$。由归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$ 且
          $\mSat{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvOr}{\iftag{probOr}{练习。}{\indcase{!A}{(!B \lor
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat{M}{!B}[w]$ 或
          $\mSat{M}{!C}[w]$。由归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$ 或
          $\mSat{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvIf}{\iftag{probIf}{练习。}{\indcase{!A}{(!B \lif
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat/{M}{!B}[w]$ 或
          $\mSat{M}{!C}[w]$。由归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat/{M^*}{!B}[{[w]}]$ 或
          $\mSat{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvIff}{\iftag{probIff}{练习。}{\indcase{!A}{(!B \liff
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat{M}{!B}[w]$ 且
          $\mSat{M}{!C}[w]$，或 $\mSat/{M}{!B}[w]$ 且
          $\mSat/{M}{!C}[w]$。由归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$ 且
          $\mSat{M^*}{!C}[{[w]}]$，或 $\mSat/{M^*}{!B}[{[w]}]$ 且
          $\mSat/{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvBox}{\iftag{probBox}{练习。}{\indcase{!A}{\Box
          !B}{假设 $\mSat{M}{\indfrm}[w]$；为证 $\mSat{M^*}{\indfrm}[{[w]}]$，令 $v$ 满足
          $R^*[w][v]$。根据
          \olref{defn:filtration}\olref{defn:filtration-R2}，我们有
          $\mSat{M}{!B}[v]$，且由归纳假设 $\mSat{M^*}{!B}[{[v]}]$。由于 $v$ 是任意的，
          可得 $\mSat{M^*}{\indfrm}[{[w]}]$。

          反之，假设 $\mSat{M^*}{\indfrm}[{[w]}]$ 并令 $v$
          是任意满足 $Rwv$ 的世界。根据
          \olref{defn:filtration}\olref{defn:filtration-R1}，我们有
          $R^*[w][v]$，因此 $\mSat{M^*}{!B}[{[v]}]$；由归纳假设 $\mSat{M}{!B}[v]$，又因 $v$ 是任意的，故有
          $\mSat{M}{\indfrm}[w]$。}}}{}
    
    \tagitem{prvDiamond}{\iftag{probDiamond}{练习。}{\indcase{!A}{\Diamond
          !B}{假设 $\mSat{M}{\indfrm}[w]$。则存在 $v \in W$ 使得
          $Rwv$ 且 $\mSat{M}{!B}[v]$。 由归纳假设 $\mSat{M^*}{!B}[{[v]}]$，再根据
          \olref{defn:filtration}\olref{defn:filtration-R1}，我们有
          $R^*[w][v]$。因此，$\mSat{M^*}{\indfrm}[{[w]}]$。

          现在假设 $\mSat{M^*}{\indfrm}[{[w]}]$。则存在满足 $R^*[w][v]$ 的 $[v]
          \in W^*$ 使得 $\mSat{M^*}{!B}[{[v]}]$。 由
          归纳假设 $\mSat{M}{!B}[v]$。 根据
          \olref{defn:filtration}\olref{defn:filtration-R3}，我们有
          $\mSat{M}{\indfrm}[w]$。}}}{}
  \end{enumerate}
\end{proof}

\begin{prob}
  完成 \olref[nml][fil][fil]{thm:filtrations} 的证明。
\end{prob}

在模型中诸世界处为真的结论，对模型中的真值以及模型类中的有效性同样成立。

\begin{cor}
  设 $\Gamma$ 对子公式封闭。则：
  \begin{enumerate}
  \item 若 $\mModel{M^*}$ 是 $\mModel{M}$ 通过 $\Gamma$ 的过滤，则对任意 $!A \in \Gamma$：$\mSat{M}{!A}$ 当且仅当 $\mSat{M^*}{!A}$。
  \item 若 $\mClass{C}$ 是一个模型类，$\Gamma(\mClass{C})$ 是 $\mClass{C}$ 中模型的 $\Gamma$-过滤构成的类，则任何公式~$!A \in \Gamma$ 在 $\mClass{C}$ 中有效当且仅当它在 $\Gamma(\mClass{C})$ 中有效。
  \end{enumerate}
\end{cor}
