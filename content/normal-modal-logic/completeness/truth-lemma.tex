\olsection{真值引理}

典范模型~$\mModel{M^\Sigma}$ 的定义方式使得
$\mSat{M^\Sigma}{!A}[\Delta]$ 当且仅当 $!A \in \Delta$。
对于命题变元，这由 $V^\Sigma$ 的定义直接得出。
然而，我们需要验证该等价关系对所有公式均成立。
我们通过归纳法来验证。
归纳步骤涉及为包含命题算子（此时须用到 \olref[ccs]{prop:ccs-properties}）
和模态算子（此时须援引 \olref[mod]{sec} 的结果）的公式证明该等价关系。

\begin{prop}[真值引理]
  \ollabel{prop:truthlemma}
  对每个公式~$!A$，$\mSat{M^\Sigma}{!A}[\Delta]$ 当且仅当
  $!A \in \Delta$。
\end{prop}

\begin{proof}
  对~$!A$ 施归纳。
  \begin{enumerate}
    \tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\mSat/{M^\Sigma}{\lfalse}[\Delta]$
        （根据 \olref[syn][trw]{defn:mmodels}），且 $\lfalse \notin \Delta$
        （根据
        \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lfalse}）。}}{}

    \tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\mSat{M^\Sigma}{\ltrue}[\Delta]$
        （根据 \olref[syn][trw]{defn:mmodels}），且 $\ltrue \in \Delta$（根据
        \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-ltrue}）。}}{}

    \item \indcase{!A}{p}{$\mSat{M^\Sigma}{p}[\Delta]$ 当且仅当 $\Delta \in
      V^\Sigma(p)$（根据 \olref[syn][trw]{defn:mmodels}）。又，根据~$V^\Sigma$ 的定义，$\Delta
      \in V^\Sigma(p)$ 当且仅当 $p \in \Delta$。}

    \tagitem{prvNot}{\indcase{!A}{\lnot !B}{%
        \iftag{probNot}{练习。}{$\mSat{M^\Sigma}{\lnot !B}[\Delta]$
          当且仅当 $\mSat/{M^\Sigma}{!B}[\Delta]$
          （\olref[syn][trw]{defn:mmodels}），当且仅当 $!B \notin \Delta$（由
          归纳假设），当且仅当 $\lnot !B \in \Delta$（根据
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lnot}）。}}}{}

    \tagitem{prvAnd}{\indcase{!A}{!B \land !C}{%
        \iftag{probAnd}{练习。}{$\mSat{M^\Sigma}{!B \land
            !C}[\Delta]$ 当且仅当 $\mSat{M^\Sigma}{!B}[\Delta]$ 且
          $\mSat{M^\Sigma}{!C}[\Delta]$（根据
          \olref[syn][trw]{defn:mmodels}），当且仅当 $!B \in \Delta$ 且 $!C
          \in \Delta$（由归纳假设），当且仅当 $!B \land !C \in
          \Delta$（根据
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-land}）。}}}{}

    \tagitem{prvOr}{\indcase{!A}{!B \lor !C}{%
        \iftag{probOr}{练习。}{$\mSat{M^\Sigma}{!B \lor
            !C}[\Delta]$ 当且仅当 $\mSat{M^\Sigma}{!B}[\Delta]$ 或
          $\mSat{M^\Sigma}{!C}[\Delta]$（根据
          \olref[syn][trw]{defn:mmodels}），当且仅当 $!B \in \Delta$ 或 $!C
          \in \Delta$（由归纳假设），当且仅当 $!B \lor !C \in
          \Delta$（根据
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lor}）。}}}{}

    \tagitem{prvIf}{\indcase{!A}{!B \lif !C}{%
        \iftag{probIf}{练习。}{$\mSat{M^\Sigma}{!B \lif
            !C}[\Delta]$ 当且仅当 $\mSat/{M^\Sigma}{!B}[\Delta]$ 或
          $\mSat{M^\Sigma}{!C}[\Delta]$（根据
          \olref[syn][trw]{defn:mmodels}），当且仅当 $!B \notin \Delta$ 或
          $!C \in \Delta$（由归纳假设），当且仅当 $!B \lif !C
          \in \Delta$（根据
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lif}）。}}}{}

    \tagitem{prvIff}{\indcase{!A}{!B \liff !C}{%
        \iftag{probIff}{练习。}{$\mSat{M^\Sigma}{!B \liff
            !C}[\Delta]$ 当且仅当（$\mSat{M^\Sigma}{!B}[\Delta]$ 且
          $\mSat{M^\Sigma}{!C}[\Delta]$）或（$\mSat/{M^\Sigma}{!B}[\Delta]$ 且
          $\mSat/{M^\Sigma}{!C}[\Delta]$）（根据
          \olref[syn][trw]{defn:mmodels}），当且仅当（$!B \in \Delta$
          且 $!C \in \Delta$）或（$!B \notin \Delta$ 且 $!C \notin
          \Delta$）（由归纳假设），当且仅当 $!B \liff !C \in
          \Delta$（根据
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-liff}）。}}}{}

    \tagitem{prvBox}{\indcase{!A}{\Box !B}{%
        \iftag{probBox}{练习。}{首先假设
          $\mSat{M^\Sigma}{\Box !B}[\Delta]$。根据
          \olref[syn][trw]{defn:mmodels}，对每个满足 $R^\Sigma \Delta\Delta'$ 的 $\Delta'$，
          有 $\mSat{M^\Sigma}{!B}[\Delta']$。由归纳假设，对每个满足 $R^\Sigma \Delta\Delta'$ 的 $\Delta'$，有 $!B \in
          \Delta'$。根据 $R^\Sigma$ 的定义，对每个满足 $\Box^{-1} \Delta \subseteq \Delta'$ 的 $\Delta'$，有 $!B \in
          \Delta'$。根据 \olref[mod]{prop:box}，$\Box !B \in
          \Delta$。

          现在假设 $\Box !B \in \Delta$。设 $\Delta' \in W^\Sigma$ 满足 $R^\Sigma \Delta\Delta'$，即 $\Box^{-1}\Delta
          \subseteq \Delta'$。由于 $\Box !B \in \Delta$，故 $!B \in
          \Box^{-1} \Delta$。因此 $!B \in \Delta'$。由
          归纳假设，$\mSat{M^\Sigma}{!B}[\Delta']$。由于 $\Delta'$ 是满足 $R^\Sigma \Delta\Delta'$ 的任意元素，故对所有满足 $R^\Sigma
          \Delta\Delta'$ 的 $\Delta' \in W^\Sigma$，有 $\mSat{M^\Sigma}{!B}[\Delta']$。根据
          \olref[syn][trw]{defn:mmodels}，$\mSat{M^\Sigma}{\Box
            !B}[\Delta]$。}}}{}

    \tagitem{prvDiamond}{\indcase{!A}{\Diamond !B}{%
        \iftag{probDiamond}{练习。}{首先假设
          $\mSat{M^\Sigma}{\Diamond !B}[\Delta]$。根据
          \olref[syn][trw]{defn:mmodels}，存在某个满足 $R^\Sigma \Delta\Delta'$ 的 $\Delta'$，使得 $\mSat{M^\Sigma}{!B}[\Delta']$。
          由归纳假设，存在某个满足 $R^\Sigma \Delta\Delta'$ 的 $\Delta'$，使得 $!B \in \Delta'$。根据 $R^\Sigma$ 的定义，\iftag{prvBox}{存在某个满足
            $\Box^{-1} \Delta \subseteq \Delta'$ 的 $\Delta'$，使得 $!B \in \Delta'$。
            根据 \olref[mod]{prop:diamond}，}{}存在某个满足 $\Diamond \Delta' \subseteq \Delta$ 的 $\Delta'$，使得 $!B \in
          \Delta'$。由于 $!B \in \Delta'$，故 $\Diamond !B \in \Diamond
          \Delta'$，从而 $\Diamond !B \in \Delta$。

          现在假设 $\Diamond !B \in \Delta$。根据
          \olref[mod]{prop:diamond}，存在一个完备的
          $\Sigma$-一致集 $\Delta' \in W^\Sigma$，使得
          $\Diamond \Delta' \subseteq \Delta$ 且 $!B \in
          \Delta'$。\iftag{prvBox}{ 根据 \olref[mod]{lem:box-iff-diamond}，
            存在 $\Delta' \in W^\Sigma$ 使得 $\Box^{-1}
            \Delta \subseteq \Delta'$ 且 $!B \in \Delta'$。}{} 根据
          $R^\Sigma$ 的定义，有 $R^\Sigma \Delta\Delta'$，故存在 $\Delta' \in W^\Sigma$ 使得 $R^\Sigma
          \Delta\Delta'$ 且 $!B \in \Delta'$。根据
          \olref[syn][trw]{defn:mmodels}，$\mSat{M^\Sigma}{\Diamond
            !B}[\Delta]$。}}}{}
  \end{enumerate}
\end{proof}

\begin{probtag}{probFalse,probTrue,probNot,proband,probOr,probIf,probIff,probBox,probDiamond}
  完成 \olref[nml][com][tru]{prop:truthlemma} 的证明。
\end{probtag}
