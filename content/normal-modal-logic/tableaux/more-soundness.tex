\olsection{附加规则的可靠性}

我们称一个规则对一类模型是\emph{可靠的}，如果每当分析表中的一个分支在该类的某个模型中可满足，应用该规则后得到的分支在该类的某个模型中也可满足。

\begin{prop}
  \ollabel{prop:soundness-T}
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ 和 }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  对自反模型\iftag{notprvBox,notprvDiamond}{是}{都是}可靠的。
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{通过对 $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$ 应用 T$\Box$ 来扩展该分支：
  \iftag{probBox}{练习。}{这会在分支上产生一个新的带号公式~$\sFmla{\True}{!B}[\sigma]$。假设 $\mSat{M}{\Gamma}[f]$，特别地，有 $\mSat{M}{\Box
      !B}[f(\sigma)]$。由于 $R$ 是自反的，故 $Rf(\sigma)f(\sigma)$。因此，$\mSat{M}{!B}[f(\sigma)]$，即 $\mModel{M}, f$ 满足 $\sFmla{\True}{!B}[\sigma]$。}}{}
\tagitem{prvDiamond}{通过对 $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$ 应用 T$\Diamond$ 来扩展该分支：
  \iftag{probDiamond}{练习。}{这会在分支上产生一个新的带号公式~$\sFmla{\False}{!B}[\sigma]$。假设 $\mSat{M}{\Gamma}[f]$，特别地，有 $\mSat/{M}{\Diamond
      !B}[f(\sigma)]$。由于 $R$ 是自反的，故 $Rf(\sigma)f(\sigma)$。因此，$\mSat/{M}{!B}[f(\sigma)]$，即 $\mModel{M}, f$ 满足 $\sFmla{\False}{!B}[\sigma]$。}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  完成 \olref[nml][tab][msn]{prop:soundness-T} 的证明。
\end{prob}
\tagendprob

      !B}[f(\sigma.n)]$。由于 $f$ 是分支上前缀在

    有 $Rf(\sigma.n)f(\sigma)$。由 $\mSat/{M}{\Diamond !B}[f(\sigma.n)]$ 可得 $\mSat/{M}{!B}[f(\sigma)]$。因此，$\mModel{M}, f$ 满足 $\sFmla{\False}{!B}[\sigma]$。}}{}
