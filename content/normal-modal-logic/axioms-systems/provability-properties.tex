\olsection{可推导性的性质}

\begin{prop}\ollabel{prop:derivabilityfacts}
  设 $\Sigma$ 为模态系统，$\Gamma$ 为模态公式集。下列性质成立：
  \begin{enumerate}
  \item\ollabel{prop:derivabilityfacts-monotonicity} \emph{单调性}：若
    $\Gamma \Proves[\Sigma] !A$ 且 $\Gamma \subseteq \Delta$，则
    $\Delta \Proves[\Sigma] !A$；
  \item\ollabel{prop:derivabilityfacts-reflexivity}
    \emph{自反性}：若 $!A \in \Gamma$，则 $\Gamma
    \Proves[\Sigma] !A$；
  \item\ollabel{prop:derivabilityfacts-cut} \emph{切割}：若 $\Gamma
    \Proves[\Sigma] !A$ 且 $\Delta \cup \{!A\} \Proves[\Sigma] !B$，
    则 $\Gamma \cup \Delta \Proves[\Sigma] !B$；
  \item\ollabel{prop:derivabilityfacts-deduction} \emph{演绎定理}：$\Gamma \cup \{!B\}
    \Proves[\Sigma] !A$ 当且仅当 $\Gamma \Proves[\Sigma] !B \lif
      !A$；
  \item \ollabel{prop:derivabilityfacts-ruleT}% \emph{Rule T}: If
    若 $\Gamma \Proves[\Sigma] !A_1$，\dots，$\Gamma
    \Proves[\Sigma] !A_n$，且 $!A_1 \to (!A_2 \lif \cdots (!A_n \lif
    !B)\cdots)$ 为重言实例，则 $\Gamma
    \Proves[\Sigma] !B$。
  \end{enumerate}
\end{prop}

其证明是简单的练习。由 \olref{prop:derivabilityfacts} 的
\olref{prop:derivabilityfacts-ruleT} 可知，例如，若 $\Gamma \Proves[\Sigma] !A \lor !B$
且 $\Gamma \Proves[\Sigma] \lnot !A$，则 $\Gamma \Proves[\Sigma]
!B$。此外，在下文中，我们用 $\Gamma, !A \Proves[\Sigma] !B$
代替 $\Gamma \cup \{ !A\} \Proves[\Sigma] !B$。

\begin{defn}
  集合 $\Gamma$ 相对于系统 $\Sigma$ 是\emph{演绎封闭的}，当且仅当：若 $\Gamma \Proves[\Sigma] !A$，则
  $!A \in \Gamma$。
\end{defn}
