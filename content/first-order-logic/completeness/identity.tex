\olsection{等词}

\begin{explain}
前一节给出的项模型构造，足以建立一阶逻辑关于不含~$\eq$ 的集合~$\Gamma$ 的完备性。
项模型满足每一个不含~$\eq$ 的 $!A \in \Gamma^*$（从而也满足所有 $!A \in \Gamma$）。
然而，若含有~$\eq$，它便不再适用。
原因在于，此时 $\Gamma^*$ 可能包含语句~$\eq[t][t']$，但在项模型中，任何项的值就是该项本身。
因此，如果 $t$ 和 $t'$ 是不同的项，它们在项模型中的值——即 $t$ 和 $t'$——就是不同的，从而 $\eq[t][t']$ 为假。
不过，我们可以借助一种称为“商化”的构造来解决这一问题。
\end{explain}

\begin{defn}
  设 $\Gamma^*$ 是语言~$\Lang L$ 中的一个一致且完备的语句集。
  我们在~$\Lang L$ 的闭项集合上定义关系 $\approx$ 如下：
  \[
  t \approx t' \text{\quad 当且仅当 \quad} \eq[t][t'] \in \Gamma^*
  \]
\end{defn}

\begin{prop}
\ollabel{prop:approx-equiv}
关系 $\approx$ 具有以下性质：
\begin{enumerate}
\item $\approx$ 是自反的。
\item $\approx$ 是对称的。
\item  $\approx$ 是传递的。
\item 若 $t \approx t'$，$f$ 是函数符号，且 $t_1$, \dots,
  $t_{i-1}$, $t_{i+1}$, \dots, $t_n$ 是闭项，则
\[
\Atom{f}{t_1,\dots, t_{i-1}, t, t_{i+1}, \dots, t_n} \approx
\Atom{f}{t_1,\dots, t_{i-1}, t', t_{i+1}, \dots, t_n}.
\]
\item 若 $t \approx t'$，$R$ 是谓词符号，且 $t_1$, \dots,
  $t_{i-1}$, $t_{i+1}$, \dots, $t_n$ 是闭项，则
\begin{multline*}
\Atom{R}{t_1,\dots, t_{i-1}, t, t_{i+1}, \dots, t_n} \in \Gamma^* \text{ 当且仅当 } \\
\Atom{R}{t_1,\dots, t_{i-1}, t', t_{i+1}, \dots, t_n} \in \Gamma^*.
\end{multline*}
\end{enumerate}
\end{prop}

\begin{proof}
由于 $\Gamma^*$ 是一致且完备的，$\eq[t][t'] \in
\Gamma^*$ 当且仅当 $\Gamma^* \Proves \eq[t][t']$。
因此，只需证明以下几点：
\begin{enumerate}
\item 对任意闭项~$t$，$\Gamma^* \Proves \eq[t][t]$。
\item 若 $\Gamma^* \Proves \eq[t][t']$，则 $\Gamma^* \Proves \eq[t'][t]$。
\item 若 $\Gamma^* \Proves \eq[t][t']$ 且 $\Gamma^* \Proves
  \eq[t'][t'']$，则 $\Gamma^* \Proves \eq[t][t'']$。
\item 若 $\Gamma^* \Proves \eq[t][t']$，则对于每个 $n$ 元函数符号~$f$ 及闭项 $t_1$, \dots,
$t_{i-1}$, $t_{i+1}$, \dots,~$t_n$，有
\[
\Gamma^* \Proves
\eq[\Atom{f}{t_1,\dots,t_{i-1},t,t_{i+1},,\dots,t_n}][\Atom{f}{t_1,\dots,t_{i-1},t',t_{i+1},\dots,t_n}]
\]
\item 若 $\Gamma^* \Proves \eq[t][t']$ 且
$\Gamma^* \Proves
\Atom{R}{t_1,\dots,t_{i-1},t,t_{i+1},\dots,t_n}$，则对于每个 $n$ 元谓词符号~$R$ 及闭项 $t_1$, \dots,
$t_{i-1}$, $t_{i+1}$, \dots,~$t_n$，有
$\Gamma^* \Proves \Atom{R}{t_1,\dots,t_{i-1},t',t_{i+1},\dots,t_n}$
\end{enumerate}
\end{proof}

\begin{prob}
完成 \olref[fol][com][ide]{prop:approx-equiv} 的证明。
\end{prob}

\begin{defn}
设 $\Gamma^*$ 是语言~$\Lang L$ 中的一个一致且完备的集合，$t$ 是闭项，
$\approx$ 如前定义所述。则：
\[
\equivrep{t}{\approx} = \Setabs{t'}{t'\in \Trm[L], t \approx t'}
\]
且 $\equivclass{\Trm[L]}{\approx} = \Setabs{\equivrep{t}{\approx}}{t \in \Trm[L]}$。
\end{defn}

\begin{defn}
\ollabel{defn:term-model-factor}
设 $\Struct M = \Struct M(\Gamma^*)$ 是由~$\Gamma^*$ 按 \olref[mod]{defn:termmodel} 构造的项模型。
则 $\Struct{\equivclass{M}{\approx}}$ 是如下结构：
\begin{enumerate}
\item $\Domain{\equivclass{M}{\approx}} = \equivclass{\Trm[L]}{\approx}$。
\item $\Assign{c}{\equivclass{M}{\approx}} = \equivrep{c}{\approx}$
\item $\Assign{f}{\equivclass{M}{\approx}}(\equivrep{t_1}{\approx}, \dots,
  \equivrep{t_n}{\approx}) = \equivrep{\Atom{f}{t_1,\dots, t_n}}{\approx}$
\item $\tuple{\equivrep{t_1}{\approx}, \dots, \equivrep{t_n}{\approx}} \in
  \Assign{R}{\equivclass{M}{\approx}}$ 当且仅当
  $\Sat{M}{\Atom{R}{t_1,\dots, t_n}}$，即当且仅当 $\Atom{R}{t_1,\dots,
  t_n} \in \Gamma^*$。
\end{enumerate}
\end{defn}

\begin{explain}
注意，我们是通过将 $\equivclass{\Trm[L]}{\approx}$ 的元素指代为 $\equivrep{t}{\approx}$（即通过\emph{代表元}~$t \in
\equivrep{t}{\approx}$）来定义 $\Assign{f}{\equivclass{M}{\approx}}$ 和
$\Assign{R}{\equivclass{M}{\approx}}$ 的。
我们必须确保这些定义不依赖于这些代表元的选择，即对于对应同一个等价类的其他选择~$t'$（$\equivrep{t}{\approx} = \equivrep{t'}{\approx}$），定义应给出相同的结果。
例如，若 $R$ 是一元谓词符号，定义的最后一条说的是
$\equivrep{t}{\approx} \in \Assign{R}{\equivclass{M}{\approx}}$ 当且仅当
$\Sat{M}{\Atom{R}{t}}$。
如果对于某个满足 $t \approx t'$ 的其他项~$t'$，有 $\Sat/{M}{\Atom{R}{t}}$，
那么定义就会要求 $\equivrep{t'}{\approx} \notin \Assign{R}{\equivclass{M}{\approx}}$。
若 $t \approx t'$，则 $\equivrep{t}{\approx} =
\equivrep{t'}{\approx}$，但我们不可能同时有 $\equivrep{t}{\approx}
\in \Assign{R}{\equivclass{M}{\approx}}$ 和 $\equivrep{t}{\approx}
\notin \Assign{R}{\equivclass{M}{\approx}}$。
不过，\olref{prop:approx-equiv} 保证了这种情况不会发生。
\end{explain}

\begin{prop}
$\Struct{\equivclass{M}{\approx}}$ 是良定义的，即，若 $t_1$,
  \dots, $t_n$, $t_1'$, \dots, $t_n'$ 是闭项，
  且 $t_i \approx t_i'$，则
\begin{enumerate}
\item $\equivrep{\Atom{f}{t_1,\dots, t_n}}{\approx} =
    \equivrep{\Atom{f}{t_1',\dots, t_n'}}{\approx}$，即，
  \[
  \Atom{f}{t_1,\dots, t_n} \approx \Atom{f}{t_1',\dots, t_n'}
  \]
  且
\item $\Sat{M}{\Atom{R}{t_1,\dots, t_n}}$ 当且仅当
  $\Sat{M}{\Atom{R}{t_1',\dots, t_n'}}$，即，
  \[
    \Atom{R}{t_1,\dots, t_n} \in \Gamma^* \text{ 当且仅当 }
    \Atom{R}{t_1',\dots, t_n'} \in \Gamma^*.
  \]
\end{enumerate}
\end{prop}

\begin{proof}
由 \olref{prop:approx-equiv} 施归纳于~$n$ 可得。
\end{proof}

与项模型的情形一样，在证明真值引理之前，我们需要以下引理。

\begin{lem}
\ollabel{lem:val-in-termmodel-factored} 设 $\Struct M = \Struct M
 (\Gamma^*)$，则 $\Value{t}{\equivclass{M}{\approx}} = \equivrep{t}
 {\approx}$。
\end{lem}
\begin{proof}
证明与 \olref[mod]{lem:val-in-termmodel} 类似。
\end{proof}

\begin{prob}
完成~\olref[fol][com][ide]{lem:val-in-termmodel-factored} 的证明。
\end{prob}

\begin{lem}
\ollabel{lem:truth}
对所有语句~$!A$，$\Sat{\equivclass{M}{\approx}}{!A}$ 当且仅当 $!A \in \Gamma^*$。
\end{lem}

\begin{proof}
对~$!A$ 归纳，方法与 \olref[mod]{lem:truth} 的证明相同。
唯一需要额外关注的情形是 $!A \ident
\eq[t][t']$。
\begin{align*}
\Sat{\equivclass{M}{\approx}}{\eq[t][t']} & \text{ 当且仅当 } \equivrep{t}{\approx} = \equivrep{t'}{\approx}
\text{ (by definition of $\Struct{\equivclass{M}{\approx}}$)}\\
& \text{ 当且仅当 } t \approx t' \text{（根据 $\equivrep{t}{\approx}$ 的定义）}\\
& \text{ 当且仅当 } \eq[t][t'] \in \Gamma^* \text{（根据 $\approx$ 的定义）。}
\end{align*}
\end{proof}

\begin{digress}
注意，虽然 $\Struct{M(\Gamma^*)}$ 总是可数且无穷的，但 $\Struct{\equivclass{M}{\approx}}$ 却可能是有限的，因为最终可能只有有限多个等价类 $\equivrep{t}{\approx}$。
这在意料之中，因为 $\Gamma$ 可能包含要求任何满足它们的结构皆为有限的语句。
例如，$\lforall[x][\lforall[y][x =
y]]$ 是一个一致语句，但只在论域恰好包含一个元素的结构中被满足。
\end{digress}
