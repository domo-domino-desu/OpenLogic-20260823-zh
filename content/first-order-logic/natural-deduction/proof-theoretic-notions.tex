\olsection{证明论概念}

\begin{editorial}
本节汇集了自然演绎中可证性与一致性的定义。
\end{editorial}

\begin{explain}
正如我们已经定义了许多重要的语义概念（有效性、后承、可满足性）一样，我们现在来定义相应的\emph{证明论概念}。这些概念不是诉诸语句在结构中的满足性来定义的，而是诉诸某些语句能否从其他语句推导出来来定义的。这些概念是相吻合的，这是一项重要发现。它们确实相吻合，这正是\emph{可靠性}定理与\emph{完备性}定理的内容。
\end{explain}

\begin{defn}[定理]
语句~$!A$ 是\emph{定理}，如果自然演绎中存在~$!A$ 的一个推导，且其中所有假设均已被解除。若~$!A$ 是定理，则记作 $\Proves !A$；否则记作 $\Proves/ !A$。
\end{defn}

\begin{defn}[可推导性]
语句~$!A$ \emph{从}语句集~$\Gamma$ \emph{可推导}（记作 $\Gamma \Proves !A$），如果存在一个以~$!A$ 为结论的推导，且其中每个假设要么已被解除，要么属于~$\Gamma$。如果~$!A$ 从 $\Gamma$ 不可推导，则记作 $\Gamma \Proves/ !A$。
\end{defn}

\begin{defn}[一致性]
语句集~$\Gamma$ 是\emph{不一致的}，当且仅当 $\Gamma \Proves \lfalse$。如果 $\Gamma$ 不是不一致的，即 $\Gamma \Proves/ \lfalse$，则称它是\emph{一致的}。
\end{defn}

\begin{prop}[自反性]
\ollabel{prop:reflexivity}
若 $!A \in \Gamma$，则 $\Gamma \Proves !A$。
\end{prop}

\begin{proof}
仅由假设 $!A$ 本身即构成 $!A$ 的一个推导，其中每个未被解除的假设（即 $!A$）都在~$\Gamma$ 中。
\end{proof}


\begin{prop}[单调性]
\ollabel{prop:monotonicity}
若 $\Gamma \subseteq \Delta$ 且 $\Gamma \Proves !A$，则 $\Delta \Proves !A$。
\end{prop}

\begin{proof}
任何从 $\Gamma$ 推出 $!A$ 的推导，同时也是从~$\Delta$ 推出 $!A$ 的推导。
\end{proof}

\begin{prop}[传递性]
\ollabel{prop:transitivity}
若 $\Gamma \Proves !A$ 且 $\{!A\} \cup \Delta \Proves
!B$，则 $\Gamma \cup \Delta \Proves !B$。
\end{prop}

\begin{proof}
若 $\Gamma \Proves !A$，则存在 $!A$ 的一个推导~$\delta_0$，其所有未被解除的假设都在~$\Gamma$ 中。若 $\{!A\} \cup
\Delta \Proves !B$，则存在 $!B$ 的一个推导~$\delta_1$，其所有未被解除的假设都在 $\{!A\} \cup \Delta$ 中。现在考虑：
\begin{prooftree}
  \AxiomC{$\Delta, \Discharge{!A}{1}$}
  \RightLabel{$\delta_1$}
  \DeduceC{$!B$}
  \DischargeRule{\Intro{\lif}}{1}
  \UnaryInfC{$!A \lif !B$}
  \AxiomC{$\Gamma$}
  \RightLabel{$\delta_0$}
  \DeduceC{$!A$}
  \RightLabel{\Elim{\lif}}
  \BinaryInfC{$!B$}
\end{prooftree}
此时，所有未被解除的假设都在 $\Gamma \cup \Delta$ 中，因此表明 $\Gamma \cup \Delta \Proves !B$。
\end{proof}

当 $\Gamma = \{!A_1, !A_2, \ldots, !A_k\}$ 是有限集时，我们可以使用简化记号 $!A_1,!A_2,\ldots,!A_k \Proves !B$ 表示 $\Gamma \Proves !B$；特别地，$!A \Proves !B$ 表示 $\{!A\} \Proves !B$。

注意，若 $\Gamma \Proves !A$ 且 $!A \Proves !B$，则 $\Gamma \Proves !B$。由此也可得，若 $!A_1,
\dots, !A_n \Proves !B$ 且对每个~$i$ 有 $\Gamma \Proves !A_i$，则 $\Gamma \Proves !B$。

\begin{prop}
\ollabel{prop:incons}
以下命题等价。
    \begin{enumerate}
      \item \( \Gamma \) 是不一致的。
      \item \( \Gamma \Proves {!A} \) 对每个语句~\( {!A} \) 成立。
      \item \( \Gamma \Proves {!A} \) 且 \( \Gamma \Proves \lnot {!A} \) 对某个语句~\( {!A} \) 成立。
    \end{enumerate}
\end{prop}

\begin{proof}
练习。
\end{proof}

\begin{prop}[紧致性]
  \ollabel{prop:proves-compact}
  \begin{enumerate}
  \item 若 $\Gamma \Proves !A$，则存在有限子集 $\Gamma_0 \subseteq \Gamma$ 使得 $\Gamma_0 \Proves !A$。
  \item 若 $\Gamma$ 的每个有限子集都是一致的，则 $\Gamma$ 是一致的。
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item 若 $\Gamma \Proves !A$，则存在一个从 $\Gamma$ 推出 $!A$ 的推导~$\delta$。令 $\Gamma_0$ 为 $\delta$ 中所有未被解除的假设组成的集合。由于任何推导都是有限的，$\Gamma_0$ 只能包含有限多个语句。因此，$\delta$ 就是从有限子集 $\Gamma_0 \subseteq \Gamma$ 推出 $!A$ 的推导。
    \item 这是 (1) 在 $!A \ident \lfalse$ 这一特殊情况下的换质位命题。
  \end{enumerate}
\end{proof}
