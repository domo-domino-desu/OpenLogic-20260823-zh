\olsection{可推导性与量词}

\begin{explain}
完备性定理还要求分析表规则能确立本节给出的关于~$\Proves$ 的事实。
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} 如果 $c$ 是不在 $\Gamma$ 或~$!A(x)$ 中出现的常项，且 $\Gamma \Proves !A(c)$，则 $\Gamma
\Proves \lforall[x][!A(x)]$。
\end{thm}

\begin{proof}
假设 $\Gamma \Proves !A(c)$，即存在 $!B_1$, \dots, $!B_n \in \Gamma$ 以及一个针对
\begin{align*}
\{ \sFmla{\False}{!A(c)}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\intertext{的闭分析表。我们需要证明也存在一个针对}
\{ \sFmla{\False}{\lforall[x][!A(x)]}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\end{align*}
的闭分析表。取出该闭分析表，将其第一个假设替换为
$\sFmla{\False}{\lforall[x][!A(x)]}$，并在所有假设之后插入
$\sFmla{\False}{!A(c)}$。
\begin{center}
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\formula{A}(c)}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
      [\sFmla{\True}{\formula{B}_n}, 
        [][][]]]]]
\end{tableau}\qquad
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\lforall[x][\formula{A}(x)]}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
        [\sFmla{\True}{\formula{B}_n},
          [\sFmla{\False}{\formula{A}(c)}
        [][][]]]]]]
\end{tableau}
\end{center}
该分析表仍然是闭的，因为之前所有作为假设可用的语句在新分析表的顶部仍然可用。插入的这一行是正确应用 $\TRule{\False}{\lforall}$ 规则的结果，因为常项符号~$c$ 不在 $!B_1$, \dots, $!B_n$ 或 $\lforall[x][!A(x)]$ 中出现，即在新分析表中，它不出现在插入行的上方。
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}
\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$\sFmla{\False}{\lexists[x][!A(x)]}, \sFmla{\True}{!A(t)}$ 的一个闭分析表是：
  \begin{oltableau}
    [\sFmla{\False}{\lexists[x][\formula{A}(x)]}, just = \TAss
      [\sFmla{\True}{\formula{A}(t)}, just = \TAss
        [\sFmla{\False}{\formula{A}(t)}, just = {\TRule{\False}{\lexists}[1]}, close]]]
    \end{oltableau}}{}
\tagitem{prvAll}{$\sFmla{\False}{\formula{A}(t)}, \sFmla{\True}{\lforall[x][!A(x)]}, $ 的一个闭分析表是：
  \begin{oltableau}
    [\sFmla{\False}{\formula{A}(t)}, just = \TAss
      [\sFmla{\True}{\lforall[x][\formula{A}(x)]}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just = {\TRule{\True}{\lforall}[2]}, close]]]
    \end{oltableau}}{}
\end{tagenumerate}
\end{proof}
