\olsection{证明论概念}

\begin{editorial}
本节汇集了关于分析表的可证性关系与一致性的定义。
\end{editorial}

\begin{explain}
正如我们已经定义了一些重要的语义概念（有效性、语义后承、可满足性），现在我们来定义相应的\emph{证明论概念}。这些概念不是依据语句在结构中的满足性来定义的，而是依据某些封闭分析表的存在来定义的。一个重要发现是，这些概念与语义概念是重合的。这种重合正是\emph{可靠性定理与完备性定理}的内容。
\end{explain}

\begin{defn}[定理]
如果存在一个针对~$\sFmla{\False}{!A}$ 的封闭分析表，则语句~$!A$ 是一个\emph{定理}。若 $!A$ 是定理，则记作 $\Proves !A$；否则记作 $\Proves/ !A$。
\end{defn}

\begin{defn}[可推导性]
语句 $!A$ 可由语句集合~$\Gamma$ \emph{推导}，记作 $\Gamma \Proves !A$，
当且仅当存在一个有限集 $\{!B_1, \dots, !B_n\} \subseteq \Gamma$
以及一个针对集合
\[
\{
  \sFmla{\False}{!A},
  \sFmla{\True}{!B_1}, \dots,
  \sFmla{\True}{!B_n}
  \}.
\]
的封闭分析表。
如果 $!A$ 不能由 $\Gamma$ 推导，则记作 $\Gamma \Proves/ !A$。
\end{defn}

\begin{defn}[一致性]
语句集合~$\Gamma$ 是\emph{不一致的}，当且仅当存在一个有限集 $\{!B_1, \dots, !B_n\} \subseteq \Gamma$
以及一个针对集合
\[
\{
  \sFmla{\True}{!B_1}, \dots,
  \sFmla{\True}{!B_n}
  \}.
\]
的封闭分析表。
如果 $\Gamma$ 不是不一致的，则称它是\emph{一致的}。
\end{defn}

\begin{prop}[自反性]
\ollabel{prop:reflexivity}
若 $!A \in \Gamma$，则 $\Gamma \Proves !A$。
\end{prop}

\begin{proof}
若 $!A \in \Gamma$，则 $\{!A\}$ 是~$\Gamma$ 的一个有限子集，并且如下分析表
\begin{oltableau}
  [\sFmla{\False}{\formula{A}}, just = \TAss
    [\sFmla{\True}{\formula{A}}, just = \TAss,close]
  ]
\end{oltableau}
是封闭的。
\end{proof}


\begin{prop}[单调性]
\ollabel{prop:monotonicity}
若 $\Gamma \subseteq \Delta$ 且 $\Gamma \Proves !A$，则 $\Delta
\Proves !A$。
\end{prop}

\begin{proof}
$\Gamma$ 的任何有限子集也是~$\Delta$ 的有限子集。
\end{proof}

\begin{prop}[传递性]
\ollabel{prop:transitivity}
若 $\Gamma \Proves !A$ 且 $\{!A\} \cup
\Delta \Proves !B$，则 $\Gamma \cup \Delta \Proves !B$。
\end{prop}

\begin{proof}
如果 $\{!A\} \cup \Delta \Proves !B$，则存在一个有限子集 $\Delta_0 =
\{!C_1, \dots, !C_n\} \subseteq \Delta$ 使得
\begin{align*}
\{\sFmla{\False}{!B}, & \sFmla{\True}{!A}, \sFmla{\True}{!C_1},
\dots, \sFmla{\True}{!C_n}\}
\intertext{有一个封闭分析表。如果 $\Gamma \Proves !A$，那么存在 $!D_1$, \dots, $!D_m \subseteq \Gamma$ 使得}
\{\sFmla{\False}{!A}, & \sFmla{\True}{!D_1},
\dots, \sFmla{\True}{!D_m}\}
\end{align*}
有一个封闭分析表。

现在考虑以
\[
\sFmla{\False}{!B},
\sFmla{\True}{!C_1}, \dots, \sFmla{\True}{!C_n},
\sFmla{\True}{!D_1}, \dots, \sFmla{\True}{!D_m}.
\]
为假设的分析表。
对~$!A$ 应用 \Cut{} 规则。这会生成两个分支，一个包含 $\sFmla{\True}{!A}$，另一个包含 $\sFmla{\False}{!A}$。于是，
在其中一个分支上，
\[
\{\sFmla{\False}{!B}, \sFmla{\True}{!A},
\sFmla{\True}{!C_1}, \dots, \sFmla{\True}{!C_n}\}
\]
全部可用。既然存在针对这些假设的封闭分析表，我们可以将其附加到该分支；所有经过 $\sFmla{\True}{!A}$ 的分支都会闭合。在另一个分支上，
\[
\{\sFmla{\False}{!A}, \sFmla{\True}{!D_1}, \dots,
\sFmla{\True}{!D_m}\}
\]
全部可用，因此我们也能完成另一侧以得到一个封闭分析表。这表明 $\Gamma \cup \Delta \Proves !B$。
\end{proof}

注意，这特别意味着，如果 $\Gamma \Proves !A$ 且 $!A \Proves !B$，那么 $\Gamma \Proves !B$。由此也可得，如果 $!A_1,
\dots, !A_n \Proves !B$ 且对每个~$i$ 都有 $\Gamma \Proves !A_i$，那么 $\Gamma \Proves !B$。

\begin{prop}
\ollabel{prop:incons}
$\Gamma$ 是不一致的，当且仅当对每个语句~$!A$ 都有 $\Gamma \Proves !A$。
\end{prop}

\begin{proof}
练习。
\end{proof}

\begin{prop}[紧致性]
  \ollabel{prop:proves-compact}
  \begin{enumerate}
  \item 如果 $\Gamma \Proves !A$，则存在一个有限子集 $\Gamma_0 \subseteq \Gamma$ 使得 $\Gamma_0 \Proves !A$。
  \item 如果~$\Gamma$ 的每个有限子集都是一致的，那么 $\Gamma$ 是一致的。
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item 如果 $\Gamma \Proves !A$，则存在一个有限子集 $\Gamma_0 = \{!B_1, \dots, !B_n\}$ 以及一个针对
      \[
      \{\sFmla{\False}{!A}, \sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n}\}
      \]
      的封闭分析表。此分析表也表明 $\Gamma_0 \Proves !A$。
    \item 如果 $\Gamma$ 不一致，则对某个有限子集
      $\Gamma_0 = \{!B_1, \dots, !B_n\}$，存在一个针对
      \[
      \{\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n}\}
      \]
      的封闭分析表。这个封闭分析表表明 $\Gamma_0$ 是不一致的。
  \end{enumerate}
\end{proof}
