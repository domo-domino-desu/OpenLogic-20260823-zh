\olsection{语义概念}

\begin{explain}
给出一阶语言结构的定义之后，我们就可以定义语句的一些基本语义性质及语句之间的关系。
其中最简单的概念是语句的\emph{有效性}。
如果一个语句在每个结构中都被满足，则它是有效的。
有效语句是指无论其中的非逻辑符号如何解释，都能被满足的语句。
因此，有效语句也被称为\emph{逻辑真理}——它们是真的，即在任何结构中都得到满足，
因此它们为真仅取决于其中出现的逻辑符号及其句法结构，而不依赖于非逻辑符号或其解释。
\end{explain}

\begin{defn}[有效性]
语句 $!A$ 是\emph{有效的}，记作 $\Entails !A$，当且仅当对于每个结构~$\Struct M$ 都有 $\Sat{M}{!A}$。
\end{defn}

\begin{defn}[后承]
语句集 $\Gamma$ \emph{语义地后承}语句 $!A$，记作 $\Gamma
\Entails !A$，当且仅当对于每个满足 $\Sat{M}{\Gamma}$ 的结构~$\Struct M$，都有 $\Sat{M}{!A}$。
\end{defn}

\begin{defn}[可满足性]
语句集 $\Gamma$ 是\emph{可满足的}，如果存在某个结构~$\Struct M$ 使得 $\Sat{M}{\Gamma}$。
如果 $\Gamma$ 不可满足，则称之为\emph{不可满足的}。
\end{defn}

\begin{prop}
语句 $!A$ 是有效的，当且仅当对于每个语句集 $\Gamma$ 都有 $\Gamma \Entails !A$。
\end{prop}

\begin{proof}
正向：设 $!A$ 有效，且 $\Gamma$ 为一语句集。设 $\Struct M$ 是一个满足 $\Sat{M}{\Gamma}$ 的结构。由于 $!A$ 有效，故 $\Sat{M}{!A}$，因而 $\Gamma \Entails !A$。

反向（证其逆否命题）：设 $!A$ 无效，则存在结构 $\Struct M$ 使得 $\Sat/{M}{!A}$。取 $\Gamma = \{ \ltrue \}$，由于 $\ltrue$ 有效，故 $\Sat{M}{\Gamma}$。于是，存在结构 $\Struct M$ 使得 $\Sat{M}{\Gamma}$ 但 $\Sat/{M}{!A}$，因此 $\Gamma$ 不语义地后承 $!A$。
\end{proof}

\begin{prop}
\ollabel{prop:entails-unsat}
$\Gamma \Entails !A$ 当且仅当 $\Gamma \cup \{\lnot !A\}$ 不可满足。
\end{prop}

\begin{proof}
正向：假设 $\Gamma \Entails !A$，并反设存在结构 $\Struct M$ 使得 $\Sat{M}{\Gamma
  \cup \{ \lnot !A \}}$。由于 $\Sat{M}{\Gamma}$ 且 $\Gamma \Entails !A$，可得 $\Sat{M}{!A}$。同时，由 $\Sat{M}{\Gamma\cup \{ \lnot !A \}}$ 可得 $\Sat{M}{\lnot !A}$，于是既有 $\Sat{M}{!A}$ 又有 $\Sat/{M}{!A}$，矛盾。因此，不可能存在这样的结构 $\Struct M$，故 $\Gamma \cup \{ \lnot !A \}$ 不可满足。

反向：假设 $\Gamma \cup \{ \lnot !A \}$ 不可满足。则对每个结构 $\Struct M$，要么 $\Sat/{M}{\Gamma}$，要么 $\Sat{M}{!A}$。因此，对每个满足 $\Sat{M}{\Gamma}$ 的结构 $\Struct M$，都有 $\Sat{M}{!A}$，即 $\Gamma \Entails !A$。
\end{proof}

\begin{prob}
\begin{enumerate}
\item 证明：$\Gamma \Entails \bot$ 当且仅当 $\Gamma$ 不可满足。
\item 证明：$\Gamma \cup \{!A\} \Entails \bot$ 当且仅当 $\Gamma \Entails \lnot !A$。
\item 假设 $c$ 不在 $!A$ 或 $\Gamma$ 中出现。证明：$\Gamma \Entails \lforall[x][!A]$ 当且仅当 $\Gamma \Entails
  \Subst{!A}{c}{x}$。
\end{enumerate}
\end{prob}

\begin{prop}
如果 $\Gamma \subseteq \Gamma'$ 且 $\Gamma \Entails !A$，那么 $\Gamma' \Entails !A$。
\end{prop}

\begin{proof}
假设 $\Gamma \subseteq \Gamma'$ 且 $\Gamma \Entails !A$。设 $\Struct M$ 是一个满足 $\Sat{M}{\Gamma'}$ 的结构；则 $\Sat{M}{\Gamma}$，并且由于 $\Gamma \Entails !A$，可得 $\Sat{M}{!A}$。因此，只要 $\Sat{M}{\Gamma'}$ 就有 $\Sat{M}{!A}$，故 $\Gamma' \Entails !A$。
\end{proof}

\begin{thm}[语义演绎定理]
\ollabel{thm:sem-deduction}
$\Gamma \cup \{!A\} \Entails !B$ 当且仅当 $\Gamma \Entails !A \lif !B$。
\end{thm}

\begin{proof}
正向：设 $\Gamma \cup \{ !A \} \Entails !B$，并设 $\Struct M$ 是一个满足 $\Sat{M}{\Gamma}$ 的结构。如果 $\Sat{M}{!A}$，则 $\Sat{M}{\Gamma \cup \{ !A \} }$，于是由 $\Gamma \cup \{ !A \}$ 语义地后承 $!B$ 可得 $\Sat{M}{!B}$。因此，$\Sat{M}{!A \lif !B}$，故 $\Gamma \Entails !A \lif !B$。

反向：设 $\Gamma \Entails !A \lif !B$，且 $\Struct M$ 是一个满足 $\Sat{M}{\Gamma \cup \{ !A
  \}}$ 的结构。则 $\Sat{M}{\Gamma}$，于是 $\Sat{M}{!A \lif !B}$，并且由于 $\Sat{M}{!A}$，可得 $\Sat{M}{!B}$。因此，只要 $\Sat{M}{\Gamma \cup \{
  !A \} }$，就有 $\Sat{M}{!B}$，即 $\Gamma \cup \{ !A \} \Entails !B$。
\end{proof}

\begin{prop}\ollabel{prop:quant-terms}
  设 $\Struct{M}$ 为一个结构，$!A(x)$ 为一个含自由变元 $x$ 的公式，$t$ 为一个闭项。则：
  \begin{tagenumerate}{prvEx,prvAll}
    \tagitem{prvEx}{$!A(t) \Entails \lexists[x][!A(x)]$}{}
    \tagitem{prvAll}{$\lforall[x][!A(x)] \Entails !A(t)$}{}
  \end{tagenumerate}
\end{prop}

\begin{proof}
  \begin{tagenumerate}{prvEx,prvAll}
    \tagitem{prvEx}{%
      \iftag{probEx}{练习。}{假设 $\Sat{M}{!A(t)}$。设 $s$ 为一个满足 $s(x) = \Value{t}{M}$ 的变元指派。那么，因为 $!A(t)$ 是一个语句，所以 $\Sat{M}{!A(t)}[s]$。根据 \olref[ext]{prop:ext-formulas}，$\Sat{M}{!A(x)}[s]$。再根据 \olref[ass]{prop:sat-quant}，$\Sat{M}{\lexists[x][!A(x)]}$。}}{}
    \tagitem{prvAll}{%
      \iftag{probAll}{练习。}{假设 $\Sat{M}{\lforall[x][!A(x)]}$。设 $s$ 为一个满足 $s(x) = \Value{t}{M}$ 的变元指派。根据 \olref[ass]{prop:sat-quant}，$\Sat{M}{!A(x)}[s]$。根据 \olref[ext]{prop:ext-formulas}，$\Sat{M}{!A(t)}[s]$。因为 $!A(t)$ 是一个语句，所以根据 \olref[ass]{prop:sentence-sat-true}，$\Sat{M}{!A(t)}$。}}{}
  \end{tagenumerate}
\end{proof}
