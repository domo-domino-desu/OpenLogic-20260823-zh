\olsection{可推导性与一致性}

我们现在来建立可推导性关系的若干性质。这些性质本身各有独立的意义，且每一个都将在完备性定理的证明中发挥作用。

\begin{prop}\ollabel{prop:provability-contr}
  若 $\Gamma \Proves !A$ 且 $\Gamma \cup \{!A\}$ 不一致，则 $\Gamma$ 不一致。
\end{prop}

\begin{proof}
存在 $\Gamma$ 的有限子集 $\Gamma_0$ 和 $\Gamma_1$，使得 $\Log{LK}$ 可推导出相继式 $\Gamma_0 \Sequent !A$ 与 $!A, \Gamma_1 \Sequent \quad$。设 $\Gamma_0 \Sequent !A$ 的 $\Log{LK}$-推导为~$\pi_0$，$!A, \Gamma_1 \Sequent \quad$ 的 $\Log{LK}$-推导为~$\pi_1$。那么我们可以推导出：
\begin{prooftree}
\AxiomC{}
\RightLabel{$\pi_0$}
\Deduce$ \Gamma_0 \fCenter !A $
\AxiomC{}
\RightLabel{$\pi_1$}
\Deduce$!A, \Gamma_1 \fCenter $
\RightLabel{\Cut}
\BinaryInf$ \Gamma_0,\Gamma_1 \fCenter $
\end{prooftree}
由于 $\Gamma_0 \subseteq \Gamma$ 且 $\Gamma_1 \subseteq \Gamma$，有 $\Gamma_0 \cup \Gamma_1 \subseteq \Gamma$，因此 $\Gamma$~不一致。
\end{proof}

\begin{prop}
\ollabel{prop:prov-incons}
$\Gamma \Proves !A$ 当且仅当 $\Gamma \cup \{\lnot !A\}$ 不一致。
\end{prop}

\begin{proof}
首先假设 $\Gamma \Proves !A$，即存在 $\Gamma \Sequent !A$ 的一个推导~$\pi_0$。通过添加一条 $\LeftR{\lnot}$ 规则，我们得到 $\lnot !A, \Gamma \Sequent \quad$ 的一个推导，亦即 $\Gamma \cup \{\lnot !A\}$ 不一致。

若 $\Gamma \cup \{\lnot !A\}$ 不一致，则存在 $\lnot !A, \Gamma \Sequent \quad$ 的一个推导~$\pi_1$。以下是 $\Gamma \Sequent !A$ 的一个推导：
\begin{prooftree}
  \Axiom$!A \fCenter !A$
  \RightLabel{\RightR{\lnot}}
  \UnaryInf$\fCenter !A, \lnot !A$
  \AxiomC{}
  \RightLabel{$\pi_1$}
  \Deduce$\lnot !A, \Gamma \fCenter$
  \RightLabel{\Cut}
  \BinaryInf$\Gamma \fCenter !A$
\end{prooftree}
\end{proof}

\begin{prob}
证明：$\Gamma \Proves \lnot !A$ 当且仅当 $\Gamma \cup \{!A\}$ 不一致。
\end{prob}

\begin{prop}\ollabel{prop:explicit-inc}
  若 $\Gamma \Proves !A$ 且 $\lnot !A \in \Gamma$，则 $\Gamma$ 不一致。
\end{prop}

\begin{proof}
  假设 $\Gamma \Proves !A$ 且 $\lnot !A \in \Gamma$。那么存在相继式 $\Gamma_0 \Sequent !A$ 的一个推导~$\pi$。相继式 $\lnot !A, \Gamma_0 \Sequent \quad$ 也是可推导的：
  \begin{prooftree}
    \AxiomC{}
    \RightLabel{$\pi$}
    \Deduce$\Gamma_0 \fCenter !A$
    \Axiom$!A \fCenter !A$
    \RightLabel{\LeftR{\lnot}}
    \UnaryInf$\lnot !A, !A \fCenter$
    \RightLabel{\LeftR{\Exchange}}
    \UnaryInf$!A, \lnot !A \fCenter$
    \RightLabel{\Cut}
    \BinaryInf$\Gamma_0, \lnot !A \fCenter$
  \end{prooftree}
  由于 $\lnot !A \in \Gamma$ 且 $\Gamma_0 \subseteq \Gamma$，这表明 $\Gamma$ 不一致。
\end{proof}

\begin{prop}\ollabel{prop:provability-exhaustive}
  若 $\Gamma \cup \{!A\}$ 与 $\Gamma \cup \{\lnot !A\}$ 均不一致，则 $\Gamma$ 不一致。
\end{prop}

\begin{proof}
存在 $\Gamma$ 的有限子集 $\Gamma_0$ 和 $\Gamma_1$，以及 $\Log{LK}$-推导 $\pi_0$ 和 $\pi_1$，分别推导出 $!A, \Gamma_0 \Sequent \quad$ 和 $\lnot !A, \Gamma_1 \Sequent \quad$。那么我们可以推导出：
\begin{prooftree}
\AxiomC{}
\RightLabel{$\pi_0$}
\Deduce$ !A, \Gamma_0 \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$ \Gamma_0 \fCenter \lnot !A$
\AxiomC{}
\RightLabel{$\pi_1$}
\Deduce$\lnot !A, \Gamma_1 \fCenter  $
\RightLabel{\Cut}
\BinaryInf$ \Gamma_0, \Gamma_1 \fCenter $
\end{prooftree}
由于 $\Gamma_0 \subseteq \Gamma$ 且 $\Gamma_1 \subseteq \Gamma$，有 $\Gamma_0 \cup \Gamma_1 \subseteq \Gamma$。因此 $\Gamma$ 不一致。
\end{proof}
