\olsection{完备性定理}

\begin{thm}\ollabel{thm:completeness}
  若 $\Gamma \Entails !A$，则 $\Gamma \Proves !A$。
\end{thm}

\begin{proof}
  我们证明逆否命题：假设 $\Gamma \Proves/ !A$。则由
  \olref[lin]{lem:lindenbaum}，存在素集~$\Gamma^*
  \supseteq \Gamma$ 使得 $\Gamma^* \Proves/ !A$。考虑如
  \olref[mod]{defn:canonical-model} 所定义的 $\Gamma^*$ 的典范模型~$\mModel{M(\Gamma^*)}$。
  对任意 $!B \in \Gamma$，有 $\Gamma^* \Proves !B$。
  注意，$\Gamma^*(\emptyseq) = \Gamma^*$。由真值引理（\olref[tru]{lem:truth}），对所有 $!B \in \Gamma$ 有
  $\mSat{M(\Gamma^*)}{!B}[\emptyseq]$，且
  $\mSat/{M(\Gamma^*)}{!A}[\emptyseq]$。这表明 $\Gamma
  \Entails/ !A$。
\end{proof}

\begin{prob}
  证明：如果 $!A$ 仅包含命题变元、$\lor$ 和~$\land$，则 $\Entails/ !A$。
  利用此结论得出，在直觉主义逻辑中 $\lif$ 不能由 $\lor$ 和~$\land$ 定义。
\end{prob}

\begin{prob}
  利用完备性定理证明：如果 $\Proves !A \lor !B$，那么
  $\Proves !A$ 或 $\Proves !B$。（提示：假定 $\mSat/{M_1}{!A}$ 且
  $\mSat/{M_2}{!B}$，并构造一个新模型 $\mModel{M}$ 使得
  $\mSat/{M}{!A \lor !B}$。）
\end{prob}

\begin{prob}
  证明：如果 $\mModel{M}$ 是带有线性序的关系模型，那么
  $\mSat{M}{(!A \lif !B)\lor(!B \lif !A)}$。
\end{prob}
