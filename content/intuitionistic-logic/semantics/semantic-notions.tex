\olsection{语义概念}

\begin{defn}
  我们称 $!A$ \emph{在模型 $\mModel{M} = \tuple{W,R,V}$ 中为真}，记作 $\mSat{M}{!A}$，当且仅当 $\mSat{M}{!A}[w]$ 对所有 $w \in W$ 成立。
  $!A$ 是\emph{有效的}，记作 $\Entails !A$，当且仅当它在所有模型中为真。
  我们称公式集合~$\Gamma$ \emph{语义地后承}~$!A$，记作 $\Gamma \Entails !A$，当且仅当对每个模型~$\mModel{M}$ 和每个满足 $\mSat{M}{\Gamma}[w]$ 的~$w$，都有 $\mSat{M}{!A}[w]$。
\end{defn}

\begin{prop}\ollabel{prop:sat-entails}
  \begin{enumerate}
  \item\ollabel{prop:sat-entails1} 如果 $\mSat{M}{\Gamma}[w]$ 且 $\Gamma \Entails !A$，那么 $\mSat{M}{!A}[w]$。
  \item\ollabel{prop:sat-entails2} 如果 $\mSat{M}{\Gamma}$ 且 $\Gamma \Entails !A$，那么 $\mSat{M}{!A}$。
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item 假设 $\mSat{M}{\Gamma}$。因为 $\Gamma \Entails !A$，我们知道如果 $\mSat{M}{\Gamma}[w]$，那么 $\mSat{M}{!A}[w]$。由于对所有 $u \in W$ 都有 $\mSat{M}{\Gamma}[u]$，故 $\mSat{M}{\Gamma}[w]$ 成立。因此 $\mSat{M}{!A}[w]$。
  \item 由 \olref{prop:sat-entails1} 直接可得。
  \end{enumerate}
\end{proof}

\begin{defn}\ollabel{defn:restrict}
  假设 $\mModel{M}$ 是关系模型且 $w \in W$。$\mModel{M}$ 到~$w$ 的\emph{限制} $\mModel{M}_w=\tuple{W_w, R_w, V_w}$ 定义如下：
  \begin{align*}
    W_w & = \Setabs{u \in W}{Rwu},\\
    R_w  & = R \cap (W_w)^2, \text{ 以及}\\
    V_w(p) & = V(p) \cap W_w.
  \end{align*}
\end{defn}

\begin{prop}\ollabel{prop:restrict}
  $\mSat{M}{!A}[w]$ 当且仅当 $\mSat{M_w}{!A}$。
\end{prop}

\begin{prob}
  证明 \olref[int][sem][sem]{prop:restrict}。
\end{prob}

\begin{prop}
  假设对每个满足 $\mSat{M}{\Gamma}$ 的模型~$\mModel{M}$，都有 $\mSat{M}{!A}$。那么 $\Gamma \Entails !A$。
\end{prop}

\begin{proof}
  假设 $\mSat{M}{\Gamma}[w]$。对每个~$!B \in \Gamma$ 应用 \olref{prop:restrict}，可得 $\mSat{M_w}{\Gamma}$。由题设得 $\mSat{M_w}{!A}$。再次应用 \olref{prop:restrict}，可得 $\mSat{M}{!A}[w]$。
\end{proof}
