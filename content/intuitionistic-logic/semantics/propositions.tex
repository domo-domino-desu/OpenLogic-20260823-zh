\olsection{命题}

有时，为关系模型~$\mModel{M}$ 中使公式~$!A$ 为真的世界集合取一个名字会很有用。我们称其为由~$!A$ 定义的\emph{命题}，并记作 $\Prop{M}{!A}$。

\begin{defn}
  我们将函数 $V$ 归纳地扩展为函数 $\Prop{M}{!A} \colon \Frm
  \to \Pow{W}$，定义如下：
  \begin{enumerate}
  \item $\Prop{M}{\lfalse} = \emptyset$
  \item $\Prop{M}{p} = V(p)$
  \item $\Prop{M}{\lnot !B} = {}$
    \[
      \Setabs{w \in W}{\text{对每个满足~$Rww'$ 的~$w'$，都有~$w'
          \notin \Prop{M}{!B}$}}.
    \]
  \item $\Prop{M}{!B \land !C} = \Prop{M}{!B} \cap \Prop{M}{!C}$。
  \item $\Prop{M}{!B \lor !C} = \Prop{M}{!B} \cup \Prop{M}{!C}$。
  \item $\Prop{M}{!B \lif !C} = {}$
    \[
      \Setabs{w \in W}{\text{对每个满足~$Rww'$ 的~$w'$，若~$w'
          \in \Prop{M}{!B}$ 则~$w' \in \Prop{M}{!C}$}}.
    \]
  \end{enumerate}
\end{defn}

\begin{prop}
  \ollabel{prop:propositions-true}
  $\mModel{M}{!A}[w]$ 当且仅当 $w \in \Prop{M}{!A}$。
\end{prop}

\begin{proof}
  练习。
\end{proof}

\begin{prob}
  证明 \olref[int][sem][pro]{prop:propositions-true}。
\end{prob}
