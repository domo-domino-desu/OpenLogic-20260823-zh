\olsection{函数复合在~$\Th{Q}$~中可表示}

假设 $h$ 由下式定义：
\[
h(x_0,\dots,x_{l-1}) = f(g_0(x_0,\dots,x_{l-1}), \dots,
g_{k-1}(x_0,\dots,x_{l-1})).
\]
其中我们已经找到了分别表示函数 $f$ 以及 $g_0$, \dots,~$g_{k-1}$ 的公式 $!A_f, !A_{g_0}, \dots,
!A_{g_{k-1}}$。我们需要找到一个表示 $h$ 的公式~$!A_h$。

先考虑所有函数都是一元的简单情形，即考虑 $h(x) = f(g(x))$。如果 $!A_f(y, z)$ 表示~$f$，且 $!A_g(x, y)$ 表示~$g$，我们需要一个表示~$h$ 的公式~$!A_h(x, z)$。注意，$h(x) = z$ 当且仅当存在一个~$y$，使得 $z = f(y)$ 且 $y = g(x)$。（如果 $h(x) = z$，那么 $g(x)$ 就是这样的一个~$y$；如果这样的 $y$ 存在，那么由 $y = g(x)$ 和 $z = f(y)$，可得 $z = f(g(x))$。）这表明 $\lexists[y][(!A_g(x, y) \land !A_f(y,
  z))]$ 是~$!A_h(x, z)$ 的一个很好的候选公式。我们只需验证 $\Th{Q}$ 能证明相关的公式。

\begin{prop}
\ollabel{prop:rep1}
如果 $h(n) = m$，则 $\Th{Q} \Proves !A_h(\num{n}, \num{m})$。
\end{prop}

\begin{proof}
假设 $h(n) = m$，即 $f(g(n)) = m$。令 $k = g(n)$。则
\begin{align*}
  \Th{Q} & \Proves !A_g(\num{n}, \num{k})
  \intertext{由于 $!A_g$ 表示~$g$，且}
  \Th{Q} & \Proves !A_f(\num{k}, \num{m})
  \intertext{由于 $!A_f$ 表示~$f$。因此，}
  \Th{Q} & \Proves !A_g(\num{n}, \num{k}) \land !A_f(\num{k}, \num{m})
  \intertext{从而也有}
  \Th{Q} & \Proves \lexists[y][(!A_g(\num{n}, y) \land !A_f(y, \num{m}))],
\end{align*}
即 $\Th{Q} \Proves !A_h(\num{n}, \num{m})$。
\end{proof}

\begin{prop}
\ollabel{prop:rep2}
如果 $h(n) = m$，则 $\Th{Q} \Proves \lforall[z][(!A_h(\num{n}, z) \lif
  z = \num{m})]$。
\end{prop}

\begin{proof}
假设 $h(n) = m$，即 $f(g(n)) = m$。令 $k = g(n)$。则
\begin{align*}
  \Th{Q} & \Proves \lforall[y][(!A_g(\num{n}, y) \lif \eq[y][\num{k}])]
  \intertext{由于 $!A_g$ 表示~$g$，且}
  \Th{Q} & \Proves \lforall[z][(!A_f(\num{k}, z) \lif \eq[z][\num{m}])]
  \intertext{由于 $!A_f$ 表示~$f$。只需稍作逻辑推导，我们也能证明}
  \Th{Q} & \Proves \lforall[z][(\lexists[y][(!A_g(\num{n}, y) \land
      !A_f(y, z))] \lif \eq[z][\num{m}])].
\end{align*}
即 $\Th{Q} \Proves \lforall[y][(!A_h(\num n, y) \lif \eq[y][\num m])]$。
\end{proof}

同样的思想适用于 $f$ 和~$g_i$ 的元数大于~$1$ 的更复杂情形。

\begin{prop}
\ollabel{prop:rep-composition}
如果 $!A_f(y_0, \dots, y_{k-1}, z)$ 在~$\Th{Q}$ 中表示 $f(y_0, \dots, y_{k-1})$，且 $!A_{g_i}(x_0, \dots, x_{l-1}, y)$ 在~$\Th{Q}$ 中表示 $g_i(x_0, \dots, x_{l-1})$，那么
\begin{multline*}
  \lexists[y_0\dots][\lexists[y_{k-1}][(!A_{g_0}(x_0,\dots,x_{l-1},y_0) \land
      \dots \land {}]]\\
  !A_{g_{k-1}}(x_0,\dots,x_{l-1},y_{k-1}) \land !A_f(y_0,\dots,y_{k-1},z))
\end{multline*}
表示
\[
h(x_0, \dots, x_{l-1}) = f(g_0(x_0, \dots, x_{l-1}), \dots, g_{k-1}(x_0,
\dots, x_{l-1})).
\]
\end{prop}

\begin{proof}
练习。
\end{proof}
