\olsection{关系上的运算}

对关系进行修改或组合往往很有用。
在\olref[sfr][rel][ord]{prop:stricttopartial}中，我们考虑了关系的\emph{并}，
这无非是把两个关系视为有序对的集合后取并集。
类似地，在\olref[sfr][rel][ord]{prop:partialtostrict}中，
我们考虑了关系的相对差。
以下是对关系进行的一些其他运算。

\begin{defn}\ollabel{relationoperations}
设 $R$, $S$ 为关系，$A$ 为任意集合。

$R$ 的\emph{逆}是 $R^{-1} = \Setabs{\tuple{y, x}}{\tuple{x,
    y} \in R}$。

$R$ 与 $S$ 的\emph{相对积}是 $(R \mid S) =
\{\tuple{x, z} : \exists y(Rxy \land Syz)\}$。

$R$ 在 $A$ 上的\emph{限制}是 $\funrestrictionto{R}{A}= R
\cap A^2$。

$R$ 对 $A$ 的\emph{应用}是 $\funimage{R}{A} = \{y :
(\exists x \in A)Rxy\}$。
\end{defn}

\begin{ex}
令 $S \subseteq \Int^2$ 为 $\Int$ 上的后继关系，即 $S = \Setabs{\tuple{x, y} \in \Int^2}{x + 1 = y}$，使得 $Sxy$ 当且仅当 $x + 1 = y$。

$S^{-1}$ 是 $\Int$ 上的前驱关系，即 $\Setabs{\tuple{x,y}\in\Int^2}{x -1 =y}$。

$S\mid S$ 是
$ \Setabs{\tuple{x,y}\in\Int^2}{x + 2 =y}$

$\funrestrictionto{S}{\Nat}$ 是 $\Nat$ 上的后继关系。

$\funimage{S}{\{1,2,3\}}$ 是 $\{2, 3, 4\}$。
\end{ex}

\begin{defn}[传递闭包]
设 $R \subseteq A^2$ 为二元关系。

$R$ 的\emph{传递闭包}是 $R^+ = \bigcup_{0 < n \in
\Nat} R^n$，其中递归定义 $R^1 = R$ 且 $R^{n+1} = R^n
\mid R$。

$R$ 的\emph{自反传递闭包}是 $R^* = R^+ \cup
\Id{A}$。
\end{defn}

\begin{ex}
取 $\Int$ 上的后继关系 $S \subseteq \Int^2$。$S^2xy$ 当且仅当 $x + 2 = y$，$S^3xy$ 当且仅当 $x + 3 = y$，等等。因此，$S^+xy$ 当且仅当存在某个 $n \geq 1$ 使得 $x + n = y$。换言之，$S^+xy$ 当且仅当 $x < y$，而 $S^*xy$ 当且仅当 $x \le y$。
\end{ex}

\begin{prob}
证明 $R$ 的传递闭包事实上是传递的。
\end{prob}
