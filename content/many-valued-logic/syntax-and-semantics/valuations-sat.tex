\olsection{赋值与满足}

\begin{defn}[赋值]
设 $V$ 为真值集合。语言 $\Lang{L}$ 到 $V$ 的一个\emph{赋值}是一个函数~$\pAssign{v}$，它将 $V$ 中的元素指派给语言的命题变元，即 $\pAssign{v} \colon \PVar \to V$。
\end{defn}

\begin{defn}\ollabel{defn:pValue}
  给定一个到多值逻辑~$\Log L$ 的真值集合 $V$ 中的赋值~$\pAssign{v}$，归纳地定义求值函数 $\pValue{v} \colon \Frm[L] \to V$ 如下：
  \begin{enumerate}
    \item $\pValue{v}(\Obj p_n) = \pAssign{v}(\Obj p_n)$；
    \item 若 $\star$ 是 $0$ 元联结词，则 $\pValue{v}(\star)  = \tf{\star}[\Log L]$；
    \item 若 $\star$ 是 $n$ 元联结词，则 \[
      \pValue{v}(\star(!A_1, \dots, !A_n)) = \tf{\star}[\Log L]
      (\pValue{v}(!A_1), \dots, \pValue{v}(!A_n)).
    \]
  \end{enumerate}
\end{defn}

\begin{defn}[满足]
\ollabel{defn:satisfaction} 公式~$!A$ 被赋值~$\pAssign{v}$\emph{满足}，记作 $\pSat{v}{!A}[\Log L]$，当且仅当 $\pValue{v}(!A)[\Log L] \in V^+$，其中 $V^+$ 是 $\Log L$ 的指定真值集合。

我们记 $\pSat/{v}{!A}[\Log L]$ 表示“$\pSat{v}{!A}[\Log
  L]$ 不成立”。若 $\Gamma$ 为公式集合，则 $\pSat{v}{\Gamma}[\Log
  L]$ 当且仅当对每个 $!A \in \Gamma$，$\pSat{v}{!A}[\Log L]$ 均成立。
\end{defn}
