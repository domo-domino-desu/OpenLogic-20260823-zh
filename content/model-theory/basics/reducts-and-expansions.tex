\section{归约与扩张}

比较拥有共同符号的语言及其结构，往往是有用或必要的。
最常见的情况是，语言~$\Lang{L}$ 中的所有符号也是语言~$\Lang{L'}$ 的一部分，
即 $\Lang{L} \subseteq \Lang{L'}$。
此时，一个 $\Lang{L}$-结构~$\Struct{M}$ 总可以通过添加额外符号的解释，
同时保持共有符号的解释不变，从而扩张为一个 $\Lang{L'}$-结构。
另一方面，从一个 $\Lang{L'}$-结构~$\Struct{M'}$ 出发，
我们只需“忽略”那些不在~$\Lang{L}$ 中的符号的解释，
即可得到一个 $\Lang{L}$-结构。

\begin{defn}
\ollabel{defn:reduct}
设 $\Lang L \subseteq \Lang L'$，$\Struct M$ 是 $\Lang L$-结构，$\Struct M'$ 是 $\Lang L'$-结构。
$\Struct M$ 是 $\Struct M'$ 到 $\Lang L$ 的\emph{归约}，且 $\Struct M'$ 是 $\Struct M$ 到 $\Lang L'$ 的\emph{扩张}，当且仅当
\begin{enumerate}
\item $\Domain{M} = \Domain{M'}$
\item 对每个常项符号~$c \in \Lang L$，$\Assign{c}{M} =
  \Assign{c}{M'}$。
\item 对每个函数符号~$f \in \Lang L$，$\Assign{f}{M} =
  \Assign{f}{M'}$。
\item 对每个谓词符号~$P \in \Lang L$，$\Assign{P}{M} =
  \Assign{P}{M'}$。
\end{enumerate}
\end{defn}

\begin{prop}
\ollabel{prop:reduct}
若 $\Lang{L}$-结构~$\Struct{M}$ 是 $\Lang{L'}$-结构~$\Struct{M'}$ 的归约，
则对所有 $\Lang{L}$-语句~$!A$，
\[
\Sat{M}{!A} \text{ 当且仅当 } \Sat{M'}{!A}.
\]
\end{prop}

\begin{proof}
  练习。
\end{proof}

\begin{prob}
证明 \olref[mod][bas][red]{prop:reduct}。
\end{prob}

\begin{defn}
给定 $\Lang{L}$-结构 $\Struct{M}$，若 $\Lang{L'} =
\Lang{L} \cup \{P\}$ 是通过添加一个 $n$ 元谓词符号~$P$ 得到的 $\Lang{L}$ 的扩张，
且 $R \subseteq \Domain{M}^n$ 是一个 $n$ 元关系，
则我们将满足 $\Assign{P}{M'} = R$ 的 $\Struct{M}$ 的扩张~$\Struct{M'}$ 记作 $\Expan{M}{R}$。
\end{defn}
