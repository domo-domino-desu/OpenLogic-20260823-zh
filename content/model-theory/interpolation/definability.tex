\olsection{可定义性定理}

插值定理的一个重要应用是 Beth（贝特）可定义性定理。要定义一个 $n$ 元关系~$R$，我们可以给出一个带有 $n$ 个自由变元且不涉及~$R$ 的公式~$!C$。这就是利用~$!C$ 对~$R$ 的\emph{显式}定义。同样可以说，若语言包含 $n$ 元谓词符号~$P$ 的理论~$\Sigma(P)$ 包含（或至少语义地后承）一个形式化的显式定义，即
\[
\Sigma(P) \Entails \lforall[x_1][\dots
  \lforall[x_n][(\Atom{P}{x_1,\dots, x_n} \liff !C(x_1, \dots,
    x_n))]].
\]
则它\emph{显式定义}了~$P$。但是，显式定义仅仅是定义（即完全确定）关系的一种方式。一个理论也可能使得：在任何模型中，~$P$ 的解释都由语言其余部分的解释所确定。可定义性定理断言，每当一个理论以这种方式确定了~$P$ 的解释——即每当它\emph{隐式定义}了~$P$——它也就显式地定义了~$P$。

\begin{defn}
设 $\Lang{L}$ 为不包含谓词符号~$P$ 的语言。$\Lang{L} \cup \{P\}$ 的语句集合 $\Sigma(P)$ \emph{显式定义}了~$P$，当且仅当存在 $\Lang{L}$ 的公式~$!C(x_1, \dots, x_n)$ 使得
\[
\Sigma(P) \Entails \lforall[x_1][\dots
  \lforall[x_n][(\Atom{P}{x_1,\dots, x_n} \liff !C(x_1, \dots,
    x_n))]].
\]
\end{defn}

\begin{defn}
设 $\Lang{L}$ 为不包含谓词符号~$P$ 和~$P'$ 的语言。$\Lang{L} \cup \{P\}$ 的语句集合 $\Sigma(P)$ \emph{隐式定义}了 $P$，当且仅当
\[
\Sigma(P) \cup \Sigma(P') \Entails \lforall[x_1][\dots
  \lforall[x_n][(\Atom{P}{x_1,\dots, x_n} \liff \Atom{P'}{x_1,\dots,
      x_n})]],
\]
其中 $\Sigma(P')$ 是将 $\Sigma(P)$ 中的 $P$ 一致地替换为 $P'$ 得到的结果。
\end{defn}

换言之，对于任意结构 $\Struct{M}$ 和 $R, R' \subseteq
\Domain{M}^n$，若既有 $\Expan{M}{R} \Entails \Sigma(P)$ 又有 $\Expan{M}{R'} \Entails \Sigma(P')$，则 $R=R'$；其中 $\Expan{M}{R}$ 是将 $\Lang{L}$ 扩张到 $\Lang{L} \cup \{P\}$ 后所得且满足 $\Assign{P}{M'} = R$ 的结构~$\Struct{M'}$，$\Expan{M}{R'}$ 类似定义。

\begin{thm}[Beth 可定义性定理]
$\Lang{L} \cup\{P\}$-公式的集合 $\Sigma(P)$ 隐式定义 $P$，当且仅当 $\Sigma(P)$ 显式定义 $P$。
\end{thm}

\begin{proof}
若 $\Sigma(P)$ 显式定义了 $P$，则既有 \begin{align*}
  \Sigma(P) & \Entails & \lforall[x_1][\dots \lforall[x_n]
    [(\Atom{P}{x_1,\dots, x_n} \liff !C(x_1,\dots,x_n))]]\\
  \Sigma(P') & \Entails & \lforall[x_1][\dots \lforall[x_n]
    [(\Atom{P'}{x_1,\dots, x_n} \liff !C(x_1,\dots,x_n))]]
\end{align*}，结论随之成立。反之：假设 $\Sigma(P)$ 隐式定义了 $P$。首先，向 $\Lang{L}$ 中添加常项符号 $c_1$, \dots,~$c_n$。则有
\[
\Sigma(P) \cup \Sigma(P') \Entails
\Atom{P}{c_1, \dots, c_n} \to  \Atom{P'}{c_1, \dots, c_n}.
\]
根据紧致性定理，存在有限子集 $\Delta_0 \subseteq \Sigma(P)$ 和 $\Delta_1 \subseteq \Sigma(P')$ 使得
\[
\Delta_0 \cup \Delta_1 \Entails
\Atom{P}{c_1, \dots, c_n} \to \Atom{P'}{c_1, \dots, c_n}.
\]
令 $!D(P)$ 为所有满足 $!A(P) \in \Delta_0$ 或 $!A(P') \in \Delta_1$ 的语句 $!A(P)$ 的合取；令 $!D(P')$ 为所有满足 $!A(P) \in \Delta_0$ 或 $!A(P') \in \Delta_1$ 的语句 $!A(P')$ 的合取。于是有 $!D(P) \land
!D(P') \Entails \Atom{P}{c_1, \dots, c_n} \to P'c_1\dots c_n$。可以将该式重新整理，使得每个谓词符号仅出现在 $\Entails$ 的一侧：
\[
!D(P) \land \Atom{P}{c_1, \dots, c_n} \Entails
!D(P') \to \Atom{P'}{c_1, \dots, c_n}.
\]
根据 Craig（克雷格）插值定理，存在一个不包含 $P$ 或 $P'$ 的语句 $!C(c_1,\dots, c_n)$ 使得：
\begin{align*}
  !D(P) \land \Atom{P}{c_1, \dots, c_n} & \Entails !C(c_1,\dots, c_n); \\
  !C(c_1,\dots, c_n) & \Entails !D(P') \to \Atom{P'}{c_1, \dots, c_n}.
\end{align*}
由这两个语义后承中的前者可得：$!D(P) \Entails
\Atom{P}{c_1,\dots, c_n} \lif !C(c_1,\dots, c_n)$。由后者，由于 $\Lang{L} \cup \{P\}$-模型 $\Expan{M}{R}
\Entails !A(P)$ 当且仅当对应的 $\Lang{L} \cup
\{P'\}$-模型 $\Expan{M}{R} \models !A(P')$，故可得 $!C(c_1,\dots, c_n) \Entails !D(P) \lif \Atom{P}{c_1,\dots, c_n}$，进而有：
\[
!D(P) \Entails !C(c_1,\dots,c_n) \to \Atom{P}{c_1,\dots, c_n}.
\]
将两者合并，得 $!D(P) \Entails \Atom{P}{c_1,\dots, c_n}
\liff !C(c_1, \dots, c_n)$，再根据语义后承的单调性与概括规则，亦有
\[
\Sigma(P) \Entails
\lforall[x_1][\dots\lforall[x_n][(\Atom{P}{x_1,\dots, x_n} \liff
    !C(x_1,\dots, x_n))]].
\]
\end{proof}
