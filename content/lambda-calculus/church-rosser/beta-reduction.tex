\begin{lem}\ollabel{lem:one-par}
  若 $M \bredone M'$，则 $M \bredpar M'$。
\end{lem}

\begin{proof}
若 $M \bredone M'$，则存在 $x$、$N$ 和 $Q$ 使得 $M$ 为 $(\lambd[x][N])Q$，$M'$ 为 $\Subst{N}{Q}{x}$。
由 \olref[pb]{thm:refl} 知 $N \bredpar N$ 且 $Q \bredpar Q$，故根据 \olref[pb]{defn:bredpar}\olref[pb]{defn:bredpar4} 立得 $(\lambd[x][N])Q
    \bredpar \Subst{N}{Q}{x}$。
\end{proof}

\begin{lem}\ollabel{lem:par-red}
  若 $M \bredpar M'$，则 $M \bred M'$。
\end{lem}

\begin{proof} 对 $M \bredpar M'$ 的推导进行归纳。
  \begin{enumerate}
  \item 最后一条规则是 \olref[pb]{defn:bredpar1}：则 $M$ 和 $M'$
    即为 $x$，且 $x \bred x$。
  \item 最后一条规则是 \olref[pb]{defn:bredpar2}：存在 $x$, $N$, $N'$ 使得 $M$ 为 
    $\lambd[x][N]$ 且 $M'$ 为 $\lambd[x][N']$，其中
    $N \bredpar N'$。由归纳假设得 $N \bred N'$。则
    $\lambd[x][N] \bred \lambd[x][N']$（经由与 $N \bred N'$ 相同的一系列
    $\bredone$ 收缩）。
  \item 最后一条规则是 \olref[pb]{defn:bredpar3}：存在 $P$, $Q$, $P'$, $Q'$ 使得 $M$ 为 
    $PQ$ 且 $M'$ 为 $P'Q'$，其中 $P \bredpar P'$
    且 $Q \bredpar Q'$。由归纳假设得 $P \bred P'$ 且 $Q \bred
    Q'$。于是，经由归约序列 $P \bred P'$ 并继之以归约 $Q \bred Q'$，可得 $PQ \bred P'Q'$。
  \item 最后一条规则是 \olref[pb]{defn:bredpar4}：存在 $x$,
    $N$, $M'$, $Q$, $Q'$ 使得 $M$ 为
    $(\lambd[x][N])Q$ 且 $M'$ 为 $\Subst{N'}{Q'}{x}$，其中 $N \bredpar N'$ 且 $Q \bredpar Q'$。
    由归纳假设得 $Q \bred Q'$ 且 $N \bred N'$。因此，
    经由 $N \bred N'$，继之以 $Q \bred Q'$，最后将 $(\lambd[x][N'])Q'$ 收缩为
    $\Subst{N'}{Q'}{x}$，可得 $(\lambd[x][N])Q \bred \Subst{N'}{Q'}{x}$。
  \end{enumerate}
\end{proof}

\begin{lem}\ollabel{lem:str}
  $\bred$ 是包含 $\bredpar$ 的最小传递关系。
\end{lem}

\begin{proof}
  令 $\xred$ 为包含 $\bredpar$ 的最小传递关系。

  $\bred \subseteq \xred$：假设 $M \bred M'$，即 $M \ident M_1
  \bredone \dots \bredone M_k \ident M'$。根据 \olref{lem:one-par}，$M
  \ident M_1 \bredpar \dots \bredpar M_k \ident M'$。由于 $\xred$
  包含 $\bredpar$ 且是传递的，故 $M \xred M'$。

  $\xred \subseteq \bred$：假设 $M \xred M'$，即 $M
  \ident M_1 \bredpar \dots \bredpar M_k \ident M'$。根据 \olref{lem:par-red}，$M \ident M_1
  \bred \dots \bred M_k \ident M'$。由于 $\bred$ 是传递的，故 $M
  \bred M'$。
\end{proof}

\begin{thm}\ollabel{thm:cr}
  $\bred$ 满足 邱奇--罗瑟 性质。
\end{thm}

\begin{proof}
  由 \olref[dap]{thm:str}、\olref[pb]{thm:cr} 及 \olref{lem:str} 直接可得。
\end{proof}
